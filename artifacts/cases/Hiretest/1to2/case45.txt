const int MAX = 10;
int globalVar = 5;
int sum(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int localVar = 3;
    int result = sum(globalVar, localVar);
    printResult(result);
    
    // This is a single line comment
    int newVar = globalVar + localVar; // Corrected: replaced '|' with '+'
    int arr[MAX];
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i;
    }
    if (newVar > 0) {
        printf("newVar is positive\n");
    } else {
        printf("newVar is non-positive\n");
    }
    return 0;
}
