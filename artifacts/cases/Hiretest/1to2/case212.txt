const int CONST_VAL = 5;
int globalVar = 10;
int arr[5] = {1, 2, 3, 4, 5};
int add(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
    return;
}
int main(){
    int a = getint();
    int b = getint();
    int sum = add(a, b);
    printResult(sum);
    int condition = (a < CONST_VAL & b > globalVar); // 使用单独的 &，应视为非法符号
    if (condition) {
        printf("Condition met.\n");
    } else {
        printf("Condition not met.\n");
    }
    return 0;
}
