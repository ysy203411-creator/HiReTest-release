const int CONST_VAL = 10;
int globalVar = 5;
int add(int a, int b) {
    return a + b;
}
int multiply(int a, int b) {
    return a * b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int localVar = 3;
    int sum = add(globalVar, localVar);
    int product = multiply(CONST_VAL, sum);
    printResult(product);
    return 0;
}
