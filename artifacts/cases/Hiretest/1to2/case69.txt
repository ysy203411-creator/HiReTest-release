const int CONST_VALUE = 10;
int global_var = 5;
void printValue(int value) {
    printf("Value: %d\n", value);
}
int add(int a, int b) {
    return a + b;
}
int main(){
    int local_var = 3;
    int result = add(global_var, CONST_VALUE);
    
    if (result > 0) {
        printValue(result);
    } else {
        printValue(local_var);
    }
    
    // Intentional error for testing: using single & instead of &&
    if (local_var & CONST_VALUE) {
        printf("Error detected\n");
    }
    
    return 0;
}
