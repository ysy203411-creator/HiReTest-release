const int MAX = 100;
int add(int a, int b) {
    return a + b;
}
int subtract(int a, int b) {
    return a - b;
}
int multiply(int a, int b) {
    return a * b;
}
int divide(int a, int b) {
    if (b == 0) {
        printf("Error: Division by zero\n");
        return 0;
    }
    return a / b;
}
int main(){
    int x = 10;
    int y = 20;
    int z = 1; //Modify the initial value of z to avoid division by zero
    int result;
    result = add(x, y);
    printf("Addition: %d\n", result);
    result = subtract(x, y);
    printf("Subtraction: %d\n", result);
    result = multiply(x, y);
    printf("Multiplication: %d\n", result);
    result = divide(x, z); //The modified z does not cause division by zero
    printf("Division: %d\n", result);
    return 0;
}
