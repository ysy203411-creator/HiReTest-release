const int a = 10;
int b = 20;
int c[5] = {1, 2, 3};
int add(int x, int y) {
    return x + y;
}
void print_result(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int result = add(a, b);
    print_result(result);
    if (a | b) {  // 错误：使用单独的 |，应为 || 
        printf("This should be an error.\n");
    }
    if (a & b) {  // 错误：使用单独的 &，应为 && 
        printf("This should also be an error.\n");
    }
    return 0;
}
