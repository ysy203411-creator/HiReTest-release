const int MAX_SIZE = 10;
int globalArray[MAX_SIZE] = {1, 2, 3, 4, 5};
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    int localArray[5] = {6, 7, 8, 9, 10};
    int result1 = sum(globalArray, MAX_SIZE);
    int result2 = sum(localArray, 5);
    printf("Sum of globalArray: %d\n", result1);
    printf("Sum of localArray: %d\n", result2);
    return 0;
}
