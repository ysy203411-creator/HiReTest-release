const int CONST_VAL = 10;
int global_var = 5;
int array[5] = {1, 2, 3, 4, 5};
void printValue(int value) {
    printf("Value: %d\n", value);
}
int calculateSum(int[] arr) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < CONST_VAL; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int local_var = 3;
    int result = 0;
    if (local_var > 0 & global_var < 10) { // 错误：使用单独的 & 应为 &&
        result = calculateSum(array);
    } else {
        printValue(CONST_VAL);
    }
    printf("Result: %d\n", result);
    return 0;
}
