const int CONST_VAL = 10;
int global_var = 5;
int arr[3] = {1, 2, 3};
int add(int a, int b) {
    return a + b;
}
void printSum(int a, int b) {
    int sum = add(a, b);
    printf("Sum: %d\n", sum);
    return;
}
int main(){
    int local_var = 7;
    int result = add(global_var, local_var);
    printf("Result: %d\n", result);
    printSum(CONST_VAL, arr[0]);
    if (result > 10 & result < 20) { // This line contains an error (single & instead of &&)
        printf("Result is between 10 and 20\n");
    }
    return 0;
}
