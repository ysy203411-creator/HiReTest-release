const int MAX = 10;
int arr[MAX];
int sum(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int x = 5;
    int y = 3;
    int result;
    if (x & y) {  // 使用单独的 &，应视为非法符号
        result = sum(x, y);
    } else {
        result = sum(x, -y);
    }
    printResult(result);
    return 0;
}
