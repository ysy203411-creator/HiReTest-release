const int a = 10;
const int b = 20;
int main(){
    int x = 5;
    int y = 15;
    int result;
    
    //Use of illegal symbols & and |, should be identified as error
    if (x & y) {
        result = a + b;
    } else {
        result = a - b;
    }
    
    if (x | y) {
        result = a * b;
    } else {
        result = a / b;
    }
    
    printf("Result: %d\n", result);
    return 0;
}
