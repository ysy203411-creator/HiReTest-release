const int a = 5;
const int b = 10;
int c = 0;
void checkRelExp(int x, int y) {
    if (x < y) {
        printf("x is less than y\n");
    }
    if (x > y) {
        printf("x is greater than y\n");
    }
    if (x <= y) {
        printf("x is less than or equal to y\n");
    }
    if (x >= y) {
        printf("x is greater than or equal to y\n");
    }
}
int main(){
    c = a + b;
    checkRelExp(a, b);
    checkRelExp(b, a);
    checkRelExp(a, a);
    return 0;
}
