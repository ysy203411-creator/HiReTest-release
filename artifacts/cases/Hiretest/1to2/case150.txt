const int MAX = 10;
int arr[MAX];
int add(int x, int y) {
    return x + y;
}
int main(){
    int a = 5;
    int b = 10;
    int result = add(a, b);
    printf("Result: %d\n", result);
    
    if (a & b) { // This line contains an error: single & instead of &&
        printf("This should not print.\n");
    }
    
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * 2;
    }
    
    printf("Array element: %d\n", arr[5]);
    return 0;
}
