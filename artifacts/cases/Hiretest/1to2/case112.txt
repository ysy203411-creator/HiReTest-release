const int MAX = 10;
int arr[MAX];
int sum(int[] numbers, int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + numbers[i];
    }
    return total;
}
int main(){
    int result;
    int value = 5;
    if (value & 1) { // 使用单独的 &，这是一个错误
        printf("Odd\n");
    } else {
        printf("Even\n");
    }
    result = sum(arr, MAX);
    printf("Total: %d\n", result);
    return 0;
}
