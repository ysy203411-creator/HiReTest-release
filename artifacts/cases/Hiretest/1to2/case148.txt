const int MAX = 10;
int numbers[MAX];
void initArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * i;
    }
}
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initArray(numbers, MAX);
    int totalSum = sumArray(numbers, MAX);
    if (totalSum > 50 & totalSum < 100) { // 使用单独的 &，应视为非法符号
        printf("Sum is between 50 and 100: %d\n", totalSum);
    } else {
        printf("Sum is out of range: %d\n", totalSum);
    }
    return 0;
}
