const int MAX = 10;
int arr[MAX];
int sum(int[] array) {
    int total = 0;
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        total = total + array[i];
    }
    return total;
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * 2;
    }
    
    int result = sum(arr);
    
    if (result > 50 | result < 100) { // 错误：使用单独的 | 作为条件判断
        printf("Result is within the expected range.\n");
    } else {
        printf("Result is out of the expected range.\n");
    }
    
    return 0;
}
