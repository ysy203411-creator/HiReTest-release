const int MAX = 100;
int globalVar = 10;
int numbers[MAX];
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printResult(int result) {
    printf("The sum is: %d\n", result);
}
int main(){
    // Initialize the array with values
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        numbers[i] = i;
    }
    // Calculate the sum of the array
    int result = sum(numbers, MAX);
    // Print the result
    printResult(result);
    // Test multi-line comment handling
