const int MAX = 100;
int globalVar = 10;
int add(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
    return;
}
int main(){
    int localVar = 5;
    int result = add(globalVar, localVar);
    printResult(result);
    int array[MAX];
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        array[i] = i * 2;
    }
    if (result > 10 & result < 20) {  // 错误：使用单独的 &，应为 &&
        printf("Result is between 11 and 19\n");
    }
    return 0;
}
