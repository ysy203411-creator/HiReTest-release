const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
int sumArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int localArray[3] = {10, 20, 30};
    int totalSum = sumArray(globalArray, SIZE) + sumArray(localArray, 3);
    
    int value = globalArray[2]; // 修复语法错误，添加缺失的右括号
    printf("Total Sum: %d\n", totalSum);
    
    return 0;
}
