const int MAX = 10;
int numbers[MAX] = {1, 2, 3, 4, 5};
int sum = 0;
void calculateSum() {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        sum = sum + numbers[i];
    }
}
int main(){
    int result = 0;
    int flag = 1;
    if (flag & result) { // 错误：使用单独的 & 作为条件表达式
        calculateSum();
    }
    printf("Sum: %d\n", sum);
    return 0;
}
