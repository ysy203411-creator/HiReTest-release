const int N = 10;
int arr[N];
int sum;
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i;
    }
}
int calculateSum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    initializeArray(arr, N);
    sum = calculateSum(arr, N);
    printf("Sum of array elements: %d\n", sum);
    return 0;
}
