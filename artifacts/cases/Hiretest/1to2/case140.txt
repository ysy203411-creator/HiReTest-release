const int MAX = 10;
int numbers[MAX];
int sum;
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i + 1;
    }
}
int calculateSum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    initializeArray(numbers, MAX);
    sum = calculateSum(numbers, MAX);
    printf("Sum of numbers: %d\n", sum);
    
    // Intentional error: using single & in a condition
    if (sum & 1) {
        printf("Sum is odd.\n");
    } else {
        printf("Sum is even.\n");
    }
    
    return 0;
}
