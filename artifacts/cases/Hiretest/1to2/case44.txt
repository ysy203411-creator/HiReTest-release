const int SIZE = 5;
int array[SIZE] = {1, 2, 3, 4, 5};
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    int result = sum(array, SIZE);
    printf("Sum of array elements: %d\n", result);
    
    // Intentionally incorrect condition using single '|'
    if (result > 10 | result < 0) {
        printf("Result is out of expected range.\n");
    } else {
        printf("Result is within expected range.\n");
    }
    
    return 0;
}
