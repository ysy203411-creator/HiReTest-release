const int TRUE = 1;
const int FALSE = 0;
int a = 5;
int b = 10;
int c = 15;
int compare(int x, int y) {
    if (x == y) {
        return TRUE;
    } else {
        return FALSE;
    }
    return FALSE; // 修改为返回 FALSE，以符合语义和约束条件
    return 1;
}
void print_result(int result) {
    if (result == TRUE) {
        printf("Equal\n");
    } else {
        printf("Not Equal\n");
    }
}
int main(){
    int result1 = compare(a, b);
    print_result(result1);
    int result2 = compare(b, b);
    print_result(result2);
    int result3 = compare(c, a);
    print_result(result3);
    return 0;
}
