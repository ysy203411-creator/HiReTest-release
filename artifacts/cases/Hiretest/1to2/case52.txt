const int CONST_VAL = 10;
int global_var = 5;
int add(int a, int b) {
    return a + b;
}
int main(){
    int local_var = 3;
    int result;
    int arr[5] = {1, 2, 3, 4, 5};
    
    // 使用单独的 &，这是一个 a 类错误
    if (local_var & global_var) {
        result = add(CONST_VAL, arr[2]);
    } else {
        result = add(global_var, arr[4]);
    }
    
    printf("Result: %d\n", result);
    return 0;
}
