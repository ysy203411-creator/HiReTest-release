const int CONST_VAL = 5;
int globalVar = CONST_VAL;
int arr[3] = {1, 2, 3};
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < 3; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int computeSum(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 10;
    int sum = computeSum(globalVar, localVar);
    printf("Sum: %d\n", sum);
    int newArr[3] = {CONST_VAL, globalVar, localVar};
    printArray(newArr);
    return 0;
}
