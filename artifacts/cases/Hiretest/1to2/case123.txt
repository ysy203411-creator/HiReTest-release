const int MAX = 10;
int globalArray[MAX];
void fillArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * i;
    }
}
int sumArray(int arr[], int size) { // 修改了这里，去掉了const
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    fillArray(globalArray, MAX);
    int total = sumArray(globalArray, MAX);
    printf("Sum of squares: %d\n", total);
    return 0;
}
