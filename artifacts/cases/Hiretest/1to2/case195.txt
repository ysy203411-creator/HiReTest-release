const int N = 10;
int arr[N];
int sum(int a[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + a[i];
    }
    return total;
}
int main(){
    int i = 0;
    for (i = 0; i < N; i = i + 1) {
        arr[i] = i;
    }
    int result = sum(arr, N);
    printf("Sum: %d\n", result);
    return 0;
}
