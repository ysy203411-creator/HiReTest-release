const int a = 5;
int b = 10;
int add(int x, int y) {
    return x + y;
}
int main(){
    int result;
    result = add(a, b);
    printf("Result: %d\n", result);  // 添加缺少的分号
    if (result > 10) {
        printf("Greater than 10\n");
    } else {
        printf("Less or equal to 10\n");
    }
    int i = 0;
    for (i = 0; i < 5; i = i + 1) {
        printf("Loop %d\n", i);
    }
    return 0;
}
