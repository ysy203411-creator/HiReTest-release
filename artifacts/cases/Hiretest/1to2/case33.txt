const int a = 5;
const int b = 10;
int add(int x, int y) {
    return x + y;
}
int main(){
    int result1 = add(a, b);
    int result2 = add((a + b), (b - a));
    int result3 = add(a, (b * (a + 1)));
    
    if (result1 == 15) {
        printf("Result1 is correct: %d\n", result1);
    }
    
    if (result2 == 20) {
        printf("Result2 is correct: %d\n", result2);
    }
    
    if (result3 == 55) {
        printf("Result3 is correct: %d\n", result3);
    }
    
    return 0;
}
