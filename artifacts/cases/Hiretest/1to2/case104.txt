const int a = 5;
const int b = 10;
int add(int x, int y) {
    return x + y;
}
int main(){
    int result;
    result = add(a, b);
    if (result & 1) {  // 错误：单独的 '&' 应该是 '&&'
        printf("Odd\n");
    } else {
        printf("Even\n");
    }
    if (result | 0) {  // 错误：单独的 '|' 应该是 '||'
        printf("Non-zero\n");
    } else {
        printf("Zero\n");
    }
    return 0;
}
