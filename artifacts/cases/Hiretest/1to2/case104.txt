const int a = 5;
const int b = 10;
int add(int x, int y) {
    return x + y;
}
int main(){
    int result;
    result = add(a, b);
    if (result & 1) {  //Error: standalone '&' should be '&&'
        printf("Odd\n");
    } else {
        printf("Even\n");
    }
    if (result | 0) {  //Error: standalone '|' should be '||'
        printf("Non-zero\n");
    } else {
        printf("Zero\n");
    }
    return 0;
}
