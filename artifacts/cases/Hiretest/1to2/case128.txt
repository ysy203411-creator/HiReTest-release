const int MAX = 100;
int arr[MAX];
int sum(int arr[], int n) {
    int total = 0;
    int i = 0;
    for (i = 0; i < n; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printResult(int result) {
    printf("The sum is: %d\n", result);
    return;
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i;
    }
    int result = sum(arr, MAX);
    printResult(result);
    return 0;
}
