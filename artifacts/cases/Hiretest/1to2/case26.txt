int add(int x, int y) {
    return x + y;
}

    printf("Result: %d\n", result);
}

    int x = getint();
    int y = getint();

    printResult(result);
    if (x & y) {  // 使用单独的 & 作为条件表达式，触发 a 类错误

    }
    return 0;
}
