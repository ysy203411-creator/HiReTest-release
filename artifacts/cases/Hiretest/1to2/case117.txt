const int MAX_LENGTH = 100;
int arr[MAX_LENGTH];
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int total) {
    printf("Sum: %d\n", total);
}
int main(){
    int length = 5;
    arr[0] = 1;
    arr[1] = 2;
    arr[2] = 3;
    arr[3] = 4;
    arr[4] = 5;
    
    int total = sum(arr, length);
    printSum(total);
    
    // Test illegal single '&' in condition expression
    if (total & 1) {
        printf("Total is odd\n");
    } else {
        printf("Total is even\n");
    }
    
    return 0;
}
