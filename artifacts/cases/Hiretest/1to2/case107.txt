const int MAX = 10;
int arr[MAX];
int compute(int x) {
    return x * x;
}
int main(){
    int a = 5;
    int b = 10;
    int result;
    if (a & b) {  // Use of a single & should be considered an illegal symbol
        result = compute(a);
    } else {
        result = compute(b);
    }
    printf("Result: %d\n", result);
    return 0;
}
