const int MAX = 10;
int arr[MAX];
void process(int[] data) {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        data[i] = i * i;
    }
}
int main(){
    int result = 0;
    process(arr);
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        if (i & 1) {  // 使用单独的 &，应视为非法符号
            printf("%d\n", arr[i]);
        }
    }
    return 0;
}
