const int CONST_VAL = 10;
int global_var = 5;
int add(int a, int b) {
    return a + b;
}
int multiply(int a, int b) {
    return a * b;
}
int main(){
    int result1 = add(CONST_VAL, global_var);
    int result2 = multiply(result1, 2);
    printf("Result1: %d\n", result1);
    printf("Result2: %d\n", result2);
    return 0;
}
