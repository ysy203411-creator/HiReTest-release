const int MAX = 10;
int arr[MAX];
int add(int a, int b) {
    return a + b;
}
void printSum(int x, int y) {
    int sum = add(x, y);
    printf("Sum: %d\n", sum);
}
int main(){
    int a = 5;
    int b = 3;
    printSum(a, b);
    if (a & b) { // 使用单独的 &
        printf("Bitwise AND result is non-zero.\n");
    }
    return 0;
}
