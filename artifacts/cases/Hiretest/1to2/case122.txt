const int CONST_A = 5;
int globalVar = 10;
int sum(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 3;
    int result = sum(globalVar, localVar);
    
    if (result & 1) { // 错误：使用单独的 &
        printf("Result is odd: %d\n", result);
    } else {
        printf("Result is even: %d\n", result);
    }
    return 0;
}
