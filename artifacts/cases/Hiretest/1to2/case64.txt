const int CONST_ONE = 1;
const int CONST_TWO = 2;

int add(int a, int b) {
    return a + b;
}
void printSum(int a, int b) {
    int sum = add(a, b);
    printf("Sum: %d\n", sum);
}

    int localVar = CONST_ONE;
    int result = add(globalVar, localVar);
    printSum(result, CONST_TWO);

        printf("This should not compile due to single '&'\n");
    }

}
