const int MAX = 10;
int globalArray[MAX];
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    int localArray[MAX];
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        localArray[i] = i;
    }
    int result = sum(localArray, MAX);
    printf("The sum is: %d\n", result);
    return 0;
}
