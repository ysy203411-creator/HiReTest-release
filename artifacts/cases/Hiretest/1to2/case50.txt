const int MAX_SIZE = 10;
int numbers[MAX_SIZE];
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int arr[], int size) {
    int total = sum(arr, size);
    printf("Sum: %d\n", total);
}
int main(){
    int i = 0;
    for (i = 0; i < MAX_SIZE; i = i + 1) {
        numbers[i] = i * 2;
    }
    printSum(numbers, MAX_SIZE);
    int a = 3;
    int b = 5;
    if (a & b) { // Use of a single & should be considered an illegal symbol
        printf("This line contains an error.\n");
    }
    return 0;
}
