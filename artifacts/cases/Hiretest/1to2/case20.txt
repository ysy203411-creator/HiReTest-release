const int MAX_LENGTH = 100;
int globalVar = 5;
int add(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result is: %d\n", result);
    return;
}
int main(){
    int num1 = 10;
    int num2 = 20;
    int sum = add(num1, num2);
    
    printResult(sum);
    int array[MAX_LENGTH];
    int i = 0;
    for (i = 0; i < MAX_LENGTH; i = i + 1) {
        array[i] = i;
    }
    int check = (globalVar > 0) & (num1 < num2); // 使用单独的 &，应视为非法符号
    printf("Check value: %d\n", check);
    return 0;
}
