const int MAX_SIZE = 100;
int globalArray[MAX_SIZE];
int globalVar = 10;
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int sum(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 5;
    int result = sum(globalVar, localVar);
    printf("Sum: %d\n", result);
    int i = 0;
    for (i = 0; i < MAX_SIZE; i = i + 1) {
        globalArray[i] = i;
    }
    printArray(globalArray, MAX_SIZE);
    if (result > 0 & result < 20) { // 错误：使用单独的 & 符号
        printf("Result is within range.\n");
    }
    return 0;
}
