}
int main(){
    int testChar = 39; // ASCII value of '\''
    printChar(testChar);
    int localVar = 10;
    int localArray[3];

    localArray[1] = 2;
    localArray[2] = 3;
    if (globalVar == 0) {
        globalVar = localArray[0];
    }

    for (i = 0; i < 5; i = i + 1) {
        printf("%d ", globalArray[i]);
    }
