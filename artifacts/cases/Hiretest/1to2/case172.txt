const int CONST_VAL = 10;
int globalVar = 5;
int add(int a, int b) {
    return a + b;
}
void printSum(int a, int b) {
    printf("Sum: %d\n", add(a, b));
}
int main(){
    int localVar = 3;
    int result = add(globalVar, localVar);
    printf("Result: %d\n", result);
    printSum(CONST_VAL, result);
    int unaryTest = -localVar + !result;
    printf("Unary Test: %d\n", unaryTest);
    return 0;
}
