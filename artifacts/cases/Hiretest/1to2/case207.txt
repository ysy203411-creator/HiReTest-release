const int MAX = 100;
int globalVar = 10;
int array[MAX];
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int computeSum(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 5;
    int sum = computeSum(globalVar, localVar);
    printf("Sum: %d\n", sum);
    int i = 0;
    for (i = 0; i < 10; i = i + 1) {
        array[i] = i * i;
    }
    printArray(array, 10);
    if (1 & 0) { // Move the illegal use of '&' into the if condition
        printf("This should not print.\n");
    }
    return 0;
}
