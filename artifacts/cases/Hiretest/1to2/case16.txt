const int MAX_SIZE = 10;
int globalArray[MAX_SIZE] = {1, 2, 3};
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    const int LOCAL_CONST = 5;
    int localVar = LOCAL_CONST + 3;
    int localArray[3] = {localVar, LOCAL_CONST, globalArray[2]};
    
    printArray(globalArray, MAX_SIZE);
    printArray(localArray, 3);
    int condition = 1 || 0; // 修改：使用了合法的 ||
    if (condition) {
        printf("Condition is true.\n");
    } else {
        printf("Condition is false.\n");
    }
    return 0;
}
