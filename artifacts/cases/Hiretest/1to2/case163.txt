const int MAX = 100;
int globalVar = 10;
int array[5] = {1, 2, 3};
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int add(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 5;
    int result = add(globalVar, localVar);
    printf("Result: %d\n", result);
    printArray(array, 5);
    return 0;
}
