const int MAX = 100;
int globalVar = 10;
int array[MAX];
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i;
    }
}
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(array, MAX);
    int total = sumArray(array, MAX);
    if (total > 0 & total < 5000) { // Intentional single '&' to test error detection
        printf("Total is within expected range: %d\n", total);
    } else {
        printf("Total is out of range: %d\n", total);
    }
    return 0;
}
