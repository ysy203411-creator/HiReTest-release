const int CONST_VAL = 10;
int globalVar = 5;
int add(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 3;
    int result;
    result = add(CONST_VAL, (globalVar + localVar)); //Use parentheses expression
    printf("Result: %d\n", result);
    if ((result > 10) & (result < 20)) { //Use separate & to trigger error
        printf("In range\n");
    } else {
        printf("Out of range\n");
    }
    return 0;
}
