const int MAX_LENGTH = 10;
int globalArray[MAX_LENGTH] = {1, 2, 3, 4, 5};
void processArray(int arr[], int length) {
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int localArray[MAX_LENGTH] = {6, 7, 8, 9, 10};
    processArray(globalArray, MAX_LENGTH);
    processArray(localArray, MAX_LENGTH);
    return 0;
}
