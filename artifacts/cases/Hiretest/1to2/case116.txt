int add(int x, int y) {
    return x + y;
}
void printSum(int arr[], int size) {  // 修改形参类型为 int arr[]
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    printf("Sum: %d\n", sum);
}
