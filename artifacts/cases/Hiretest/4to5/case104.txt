const int MAX_LENGTH = 100;
int buffer[MAX_LENGTH];
int main(){
    int index = 0;
    int ch;
    int i = 0;
    while (i < MAX_LENGTH) {
        ch = getint();
        if (ch == 10) { // ASCII for newline
            break;
        }
        buffer[i] = ch;
        index = i + 1;
        i = i + 1;
    }
    i = 0;
    while (i < index) {
        printf("%d ", buffer[i]);
        i = i + 1;
    }
    return 0;
}
