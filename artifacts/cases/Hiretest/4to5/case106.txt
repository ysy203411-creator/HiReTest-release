const int CONST_VALUE = 10;
int globalVar = 5;
void modifyArray(int arr[]) {
    arr[0] = arr[0] + CONST_VALUE; // 使用常量值进行计算
}
int main(){
    int arr[5];
    int input;
    input = getint();
    arr[0] = input;
    modifyArray(arr);
    printf("%d\n", arr[0]); // 输出计算后的数组元素
    printf("%d\n", globalVar); // 输出全局变量
    return 0;
}
