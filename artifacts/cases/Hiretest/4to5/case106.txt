const int CONST_VALUE = 10;
int globalVar = 5;
void modifyArray(int arr[]) {
    arr[0] = arr[0] + CONST_VALUE; //Perform calculations using constant values
}
int main(){
    int arr[5];
    int input;
    input = getint();
    arr[0] = input;
    modifyArray(arr);
    printf("%d\n", arr[0]); //Output the array elements after calculation
    printf("%d\n", globalVar); //Output global variables
    return 0;
}
