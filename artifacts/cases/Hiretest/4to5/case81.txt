const int size = 100;

int main() {
    int count = 0;
    int i = 0;
    int str[size]; // 使用int数组替代字符数组

    // 读入最多 size 个整数到数组中
    i = 0;
    while (i < size) {
        str[i] = getint();
        i = i + 1;
    }

    // 统计值为 10 的元素个数
    i = 0;
    while (i < size) {
        if (str[i] == 10) {
            count = count + 1;
        }
        i = i + 1;
    }

    printf("Number of newline characters: %d\n", count);
    return 0;
}
