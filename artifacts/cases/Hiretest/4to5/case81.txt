const int size = 100;

int main() {
    int count = 0;
    int i = 0;
    int str[size]; //Use int array instead of char array

    //Read up to size integers into the array
    i = 0;
    while (i < size) {
        str[i] = getint();
        i = i + 1;
    }

    //Count the number of elements with value 10
    i = 0;
    while (i < size) {
        if (str[i] == 10) {
            count = count + 1;
        }
        i = i + 1;
    }

    printf("Number of newline characters: %d\n", count);
    return 0;
}
