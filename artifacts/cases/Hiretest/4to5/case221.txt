const int SIZE = 5;
int globalArray[SIZE];

void initializeArray(int arr[]) {
    int i = 0;
    //Use for statement instead of while statement
    for (i = 0; i < SIZE; i = i + 1) {
        arr[i] = i * 2;
    }
}

void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main() {
    int localArray[SIZE];

    initializeArray(localArray);
    printArray(localArray);

    initializeArray(globalArray);
    printArray(globalArray);

    return 0;
}
