const int MAX = 10;
int arr[MAX];
int sum(int a, int b) {
    return a + b;
}
void printArray(int arr[], int size) {
    int i = 0;
    while (i < size) {
        printf("%d ", arr[i]);
        i = i + 1;
    }
    printf("\n");
}
int main(){
    int total = 0;
    int i = 0;
    while (i < MAX) {
        arr[i] = getint();
        total = sum(total, arr[i]);
        i = i + 1;
    }
    printf("Total: %d\n", total);
    printArray(arr, MAX);
    return 0;
}
