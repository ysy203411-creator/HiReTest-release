const int SIZE = 5;
const int arr[SIZE] = {1, 2, 3, 4, 5};

int main() {
    int sum = 0;
    int i = 0;

    /*While is also not allowed here, so it is replaced with an equivalent sequential expansion to avoid loop statements*/
    sum = sum + arr[i];  /* i = 0 */
    i = i + 1;
    sum = sum + arr[i];  /* i = 1 */
    i = i + 1;
    sum = sum + arr[i];  /* i = 2 */
    i = i + 1;
    sum = sum + arr[i];  /* i = 3 */
    i = i + 1;
    sum = sum + arr[i];  /* i = 4 */

    printf("Sum of array elements: %d\n", sum);
    return 0;
}
