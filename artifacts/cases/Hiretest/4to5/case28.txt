const int LENGTH = 5;
int globalArray[LENGTH];
void initializeArray(int arr[]) {
    int i = 0;
    for (i = 0; i < LENGTH; i = i + 1) {
        arr[i] = i * 2;
    }
}
int sumArray(int arr[]) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < LENGTH; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(globalArray);
    int result = sumArray(globalArray);
    printf("Sum of array elements: %d\n", result);
    return 0;
}
