const int SIZE = 5;
int globalIntArray[SIZE];

void initializeArrays() {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        globalIntArray[i] = i;
    }
}

void printArrays() {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("Int: %d\n", globalIntArray[i]);
    }
}

int main() {
    initializeArrays();
    printArrays();
    return 0;
}
