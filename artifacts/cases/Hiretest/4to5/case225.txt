const int CONST_VAL = 10;
int globalVar;
int arr[5] = {1, 2, 3, 4, 5};

void processForLoop(int arr[]) {
    int i = 0;
    while (i < 5) {
        arr[i] = arr[i] + CONST_VAL;
        i = i + 1;
    }
}

int main() {
    globalVar = getint();
    processForLoop(arr);
    int i = 0;
    while (i < 5) {
        printf("%d\n", arr[i]);
        i = i + 1;
    }
    return 0;
}
