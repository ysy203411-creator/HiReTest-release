const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
void processArray(int arr[]) {
    int i = 0;
    while (i < SIZE) {
        printf("%d ", arr[i]);
        i = i + 1;
    }
    printf("\n");
}
int main(){
    int localArray[SIZE] = {6, 7, 8, 9, 10};
    processArray(globalArray);
    processArray(localArray);
    return 0;
}
