const int a = 5;
const int b = 10;
int result;

int compare(int x, int y) {
    int diff = x - y;
    int sign = (diff / (diff + 1)) - (diff / (diff - 1));
    /* 对于 x < y 时 diff 为负，sign 结果为 1；
       对于 x >= y 时 diff 为非负，sign 结果为 0。
       这里仅作为复杂表达式测试使用。 */
    return sign;
}

int main() {
    int input1 = getint();
    int input2 = getint();
    result = compare(input1, input2);
    printf("Comparison result: %d\n", result);
    result = compare(a, b);
    printf("Comparison result with constants: %d\n", result);
    return 0;
}
