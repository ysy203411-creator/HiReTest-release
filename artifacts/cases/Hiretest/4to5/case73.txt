const int a = 5;
const int b = 10;
int result;

int compare(int x, int y) {
    int diff = x - y;
    int sign = (diff / (diff + 1)) - (diff / (diff - 1));
    /*For x < y, diff is negative, sign result is 1;
       For x >= y, diff is non-negative, sign result is 0.
       This is only for testing complex expressions.*/
    return sign;
}

int main() {
    int input1 = getint();
    int input2 = getint();
    result = compare(input1, input2);
    printf("Comparison result: %d\n", result);
    result = compare(a, b);
    printf("Comparison result with constants: %d\n", result);
    return 0;
}
