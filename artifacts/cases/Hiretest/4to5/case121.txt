int a = 5;
int b = 10;
int result;

int compare(int x, int y) {
    return x - y;
}

int main() {
    int diff;

    diff = compare(a, b);          // a - b
    printf("a - b: %d\n", diff);

    diff = compare(b, a);          // b - a
    printf("b - a: %d\n", diff);

    diff = compare(a, 0);          // a - 0
    printf("a - 0: %d\n", diff);

    diff = compare(b, 0);          // b - 0
    printf("b - 0: %d\n", diff);

    diff = compare(a + b, a - b);  // (a + b) - (a - b)
    printf("(a + b) - (a - b): %d\n", diff);

    /* 去掉逻辑非，直接使用比较结果 */
    result = compare(a, 0);        // a - 0
    printf("a - 0 (no !): %d\n", result);

    result = compare(b, 0);        // b - 0
    printf("b - 0 (no !): %d\n", result);

    return 0;
}
