int globalVar;
const int constVar = 10;
int array[5];
void checkIdent() {
    if (globalVar == 0) {
        printf("globalVar is zero\\n");
    }
    if (constVar > 5) {
        printf("constVar is greater than 5\\n");
    }
    if (array[0] == 0) {
        printf("array[0] is zero\\n");
    }
}
int main(){
    globalVar = getint();
    array[0] = getint();
    checkIdent();
    return 0;
}
