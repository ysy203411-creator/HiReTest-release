const int CONST_VAL = 10;
int globalVar;
int array[5] = {1, 2, 3, 4, 5};
void printArray(int arr[], int size) {
    for (int i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int localVar = CONST_VAL;
    globalVar = getint();
    printf("Global Var: %d\n", globalVar);
    printf("Local Var: %d\n", localVar);
    printArray(array, 5);
    return 0;
}
