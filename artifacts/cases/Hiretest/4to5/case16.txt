const int TRUE = 1;
const int FALSE = 0;
int x;
int y;
int main(){
    x = getint(); //Input for controlling logical short circuit
    y = getint(); //Input for controlling logical short circuit
    if (x || (y && FALSE)) {
        printf("Short-circuit works: %d\n", TRUE);
    } else {
        printf("Short-circuit failed: %d\n", FALSE);
    }
    return 0;
}
