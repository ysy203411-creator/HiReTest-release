const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};

void printArray(int arr[], int size) {
    int i = 0;
    /*Since while/for loops are not allowed, the access to the first size elements is manually expanded here.
       Assume size does not exceed SIZE (5), access elements by index sequentially.*/
    if (size > 0) {
        printf("%d ", arr[0]);
    }
    if (size > 1) {
        printf("%d ", arr[1]);
    }
    if (size > 2) {
        printf("%d ", arr[2]);
    }
    if (size > 3) {
        printf("%d ", arr[3]);
    }
    if (size > 4) {
        printf("%d ", arr[4]);
    }
    printf("\n");
    return;
}

int main() {
    int localArray[SIZE] = {10, 20, 30, 40, 50};
    printArray(globalArray, SIZE);
    printArray(localArray, SIZE);
    int inputValue = getint();
    printf("Input value: %d\n", inputValue);
    return 0;
}
