int globalVar;
const int constVar = 10;
int array[5] = {1, 2, 3, 4, 5};
void printArray(int arr[], int size) {
    int i = 0;
    while (i < size) {
        printf("%d ", arr[i]);
        i = i + 1;
    }
    printf("\n");
}
int main(){
    int localVar = getint();
    printf("Local variable: %d\n", localVar);
    printf("Global variable: %d\n", globalVar);
    printf("Constant variable: %d\n", constVar);
    printArray(array, 5);
    return 0;
}
