const int ARRAY_SIZE = 5;
const int INIT_VALUES[5] = {1, 2, 3, 0, 0}; // 修改为一维数组，并提供全部元素

int main() {
    int i;
    int array[ARRAY_SIZE];

    for (i = 0; i < ARRAY_SIZE; i = i + 1) {
        array[i] = INIT_VALUES[i];
    }

    printf("Array values: ");
    for (i = 0; i < ARRAY_SIZE; i = i + 1) {
        printf("%d ", array[i]);
    }
    printf("\n");

    return 0;
}
