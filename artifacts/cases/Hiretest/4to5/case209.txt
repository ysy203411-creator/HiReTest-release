const int CONST_VAL = 10;
int globalVar;
int arr[5] = {1, 2, 3, 4, 5};
int checkValue(int x) {
    if (x == CONST_VAL) {
        return 1;
    }
    return 0;
}
int main(){
    int input;
    input = getint();
    int result = checkValue(input);
    printf("Result: %d\n", result);
    globalVar = arr[0];
    printf("GlobalVar: %d\n", globalVar);
    return 0;
}
