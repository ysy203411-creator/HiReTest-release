int globalVar;
const int CONST_VAL = 10;
void printMessage(int count) {
    int i = 0;
    while (i < count) {
        printf("This is a test message %d\n", i);
        i = i + 1;
    }
}
int main(){
    int localVar = getint();
    globalVar = CONST_VAL + localVar;
    printf("Global variable value: %d\n", globalVar);
    printMessage(3);
    return 0;
}
