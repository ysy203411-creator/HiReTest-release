const int SIZE = 10;
int intArray[SIZE];
int charValues[SIZE]; //Rename variable to avoid name collision

void initializeArrays() {
    int i = 0;
    /*Use while loop and arithmetic expression instead of for and character constants*/
    while (i < SIZE) {
        intArray[i] = i * 2;
        /*'a' ASCII value is 97, use integer constant here*/
        charValues[i] = 97 + i; //Rename variable
        i = i + 1;
    }
}

int getIntElement(int index) {
    /*Avoid using if and logical operators, return fixed value for out-of-range instead*/
    int result = -1;
    int i = 0;
    while (i < SIZE) {
        if (i == index) {
            result = intArray[i];
        }
        i = i + 1;
    }
    return result;
}

int getCharElement(int index) {
    int result = 0;
    int i = 0;
    while (i < SIZE) {
        if (i == index) {
            result = charValues[i]; //Rename variable
        }
        i = i + 1;
    }
    return result;
}

int main() {
    initializeArrays();
    int index = getint();
    int intElement = getIntElement(index);
    int charElement = getCharElement(index);
    printf("Integer element at %d: %d\n", index, intElement);
    printf("Character element at %d: %d\n", index, charElement);
    return 0;
}
