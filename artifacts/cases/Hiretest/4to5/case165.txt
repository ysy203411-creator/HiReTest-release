int slot;

int main() {
    int a, b;
    a = getint();
    b = getint();
    /*Use arithmetic and assignment to replace if-else, avoid using forbidden control statements*/
    slot = a - b;
    slot = slot * slot;      // slot = (a - b)^2 >= 0
    slot = slot / (a - b + slot); //If a >= b, then slot = a - b; else slot = b - a
    slot = slot + b;         // slot = max(a, b)
    printf("Slot value: %d\n", slot);
    return 0;
}
