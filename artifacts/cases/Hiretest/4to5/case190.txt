int global_var;
const int const_val = 10;
int array[5] = {1, 2, 3, 4, 5};

void processArray(int arr[]) {
    int i = 0;
    /* 使用 while 循环代替 for 循环 */
    while (i < 5) {
        arr[i] = arr[i] * 2;
        i = i + 1;
    }
}

int main() {
    int input;
    int i;
    int j;

    input = getint();

    i = 0;
    while (i < const_val) {
        global_var = global_var + i;
        i = i + 1;
    }

    printf("Global Sum: %d\n", global_var);

    processArray(array);

    j = 0;
    while (j < 5) {
        printf("Array[%d]: %d\n", j, array[j]);
        j = j + 1;
    }

    printf("Input Value: %d\n", input);
    return 0;
}
