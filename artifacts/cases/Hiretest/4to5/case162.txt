int global_var;
const int CONST_VAL = 10;
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int add(int a, int b) {
    return a + b;
}
int main(){
    int local_var = getint();
    int result = add(local_var, CONST_VAL);
    printf("Result: %d\n", result);
    int numbers[5] = {1, 2, 3, 4, 5};
    printArray(numbers, 5);
    return 0;
}
