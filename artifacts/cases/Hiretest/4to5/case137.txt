const int SIZE = 5;
int array[SIZE];

int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}

int main() {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        array[i] = getint();
    }
    int result = sum(array, SIZE);
    printf("Sum of array elements: %d\n", result);
    return 0;
}
