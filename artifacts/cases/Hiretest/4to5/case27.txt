const int SIZE = 10;
int globalArray[SIZE];

int main() {
    int i;
    int localArray1[5];
    /*Initialize a local array of length 5 with 5*/
    i = 0;
    while (i < 5) {
        localArray1[i] = 5;
        i = i + 1;
    }

    {
        int j;
        int localArray2[3];
        /*Initialize a inner local array of length 3 with 3*/
        j = 0;
        while (j < 3) {
            localArray2[j] = 3;
            j = j + 1;
        }
        /*Print the 3 elements of the inner array, should output 3 3 3*/
        printf("%d %d %d\n", localArray2[0], localArray2[1], localArray2[2]);
    }

    /*Print the 5 elements of the outer array, should output 5 5 5 5 5*/
    printf("%d %d %d %d %d\n",
           localArray1[0], localArray1[1], localArray1[2], localArray1[3], localArray1[4]);

    /*Initialize and print the first 10 elements of the global array as 10*/
    i = 0;
    while (i < SIZE) {
        globalArray[i] = 10;
        i = i + 1;
    }
    printf("%d %d %d %d %d %d %d %d %d %d\n",
           globalArray[0], globalArray[1], globalArray[2], globalArray[3], globalArray[4],
           globalArray[5], globalArray[6], globalArray[7], globalArray[8], globalArray[9]);

    return 0;
}
