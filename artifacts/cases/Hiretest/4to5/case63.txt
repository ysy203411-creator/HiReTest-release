const int TRUE = 1;
const int FALSE = 0;
int evaluate_condition(int a, int b) {
    if (a > 0 || b > 0) {
        return TRUE;
    } else {
        return FALSE;
    }
    return FALSE; //Change to a return value that meets the criteria
}
int main(){
    int x;
    int y;
    x = getint();
    y = getint();
    int result = evaluate_condition(x, y);
    printf("Condition result: %d\n", result);
    return 0;
}
