int globalVar = 0;
const int CONST_VAL = 10;
void printArray(int arr[], int size) {
    int i = 0;
    while (i < size) {
        printf("%d ", arr[i]);
        i = i + 1;
    }
    printf("\n");
}
int main(){
    int localVar = getint();
    int arr[5] = {1, 2, 3, 4, 5};
    globalVar = CONST_VAL + localVar;
    printf("Global Variable: %d\n", globalVar);
    printf("Local Variable: %d\n", localVar);
    printArray(arr, 5);
    return 0;
}
