const int CONST_VAL = 10;
int globalVar;
int array[5] = {1, 2, 3, 0, 0};  //Uninitialized elements not provided are default initialized to 0
int compute(int a, int b) {
    int result = a + b;
    return result;
}
void printArray(int arr[]) {
    int i = 0;
    //Replace while loop with equivalent if and goto statements
    loop_start:
    if (i < 5) {
        printf("%d ", arr[i]);
        i = i + 1;
        goto loop_start;
    }
    return;
}
int main(){
    int localVar = getint();
    globalVar = compute(localVar, CONST_VAL);
    printf("Result%d\n", globalVar);  //Remove disallowed characters
    if (globalVar > 10) {
        printf("GlobalVar is greater than 10\n");
    } else {
        printf("GlobalVar is 10 or less\n");
    }
    printArray(array);
    return 0;
}
