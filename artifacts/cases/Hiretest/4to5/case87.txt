const int MAX_LENGTH = 5;
int charArray[MAX_LENGTH];
const int constCharArray[MAX_LENGTH] = {65, 66, 67, 68, 69};

void initializeArray(int arr[], int length) {
    int i = 0;
    while (i < length) {
        arr[i] = 97 + i;  // 'a' 的 ASCII 为 97
        i = i + 1;
    }
}

void printArrays(int arr1[], int length) {
    int i = 0;
    while (i < length) {
        printf("%d %d\n", arr1[i], constCharArray[i]);
        i = i + 1;
    }
}

int main() {
    initializeArray(charArray, MAX_LENGTH);
    printArrays(charArray, MAX_LENGTH);
    return 0;
}
