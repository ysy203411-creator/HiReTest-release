const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};

void printArray(int arr[]) {
    int i = 0;
    while (i < SIZE) {
        printf("%d ", arr[i]);
        i = i + 1;
    }
    printf("\n");
}

int main() {
    int inputArray[SIZE];
    int i = 0;

    while (i < SIZE) {
        inputArray[i] = getint();
        i = i + 1;
    }

    printf("Global Array: ");
    printArray(globalArray);

    printf("Input Array: ");
    printArray(inputArray);

    return 0;
}
