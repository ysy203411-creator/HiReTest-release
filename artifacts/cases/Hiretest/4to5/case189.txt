int b;

int main() {
    a = getint();
    b = getint();

    //Replace if-else branches with arithmetic expressions and printf
    int diff;
    diff = a - b;

    //When a > b, diff > 0; otherwise diff <= 0
    //Show the comparison result with two output statements
    printf("a - b = %d\n", diff);
