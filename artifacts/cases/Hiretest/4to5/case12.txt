const int SIZE = 3;
int numbers[SIZE];
int sum;

int calculateSum(int[] arr) {
    int localSum = 0;
    int i = 0;
    /* 用 while 循环替代 for 循环 */
    while (i < SIZE) {
        localSum = localSum + arr[i];
        i = i + 1;
    }
    return localSum;
}

int main() {
    int i = 0;
    /* 用 while 循环替代 for 循环 */
    while (i < SIZE) {
        numbers[i] = getint();
        i = i + 1;
    }
    sum = calculateSum(numbers);
    printf("Sum: %d\n", sum);
    return 0;
}
