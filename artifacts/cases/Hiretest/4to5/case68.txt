const int a = 5;
const int b = 10;
int x;
int y;

int compare(int val1, int val2) {
    /* 
       Construct some return values directly through arithmetic expressions without using conditional statements
       Only for testing function calls and expression evaluation.
    */
    int diff = val1 - val2;
    int sum = val1 + val2;
    int prod = val1 * val2;
    int res = diff + sum + prod;
    return res;
}

int main() {
    x = getint();
    y = getint();
    int result = compare(x, y);
    printf("Comparison result: %d\n", result);
    return 0;
}
