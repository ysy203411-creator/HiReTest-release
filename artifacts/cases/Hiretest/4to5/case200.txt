const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};

int main() {
    int index;
    int value;
    int inRangeLow;
    int inRangeHigh;
    int inRange;  //1 indicates in range, 0 indicates out of range

    index = getint();

    //Compute index >= 0
    inRangeLow = 0;
    if (index >= 0) {
        inRangeLow = 1;
    }

    //Compute index < SIZE
    inRangeHigh = 0;
    if (index < SIZE) {
        inRangeHigh = 1;
    }

    //Logical AND: inRange = inRangeLow && inRangeHigh
    inRange = 0;
    if (inRangeLow) {
        if (inRangeHigh) {
            inRange = 1;
        }
    }

    //Select behavior based on inRange value (simulated via multiplication and addition)
    //When inRange == 1, fetch and print array value; otherwise print out-of-bounds message
    value = 0;
    value = globalArray[index] * inRange + value * (1 - inRange);

    //Print array value (the meaning of index and value when out of bounds is left to the test case designer)
    printf("Value at index %d is %d\n", index, value);

    return 0;
}
