const int SIZE = 5;
int array[SIZE];
void processArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        arr[i] = arr[i] * 2;
    }
}
int main(){
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        array[i] = getint();
    }
    processArray(array);
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", array[i]);
    }
    return 0;
}
