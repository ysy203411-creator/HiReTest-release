int a = 5;
int b = 10;
int c1 = 5; // 修改变量名以避免重名
int d = 0;
int isEqual(int x, int y) {
    if (x == y) {
        return 1;
    } else {
        return 0;
    }
    return 1;
}
int main(){
    int result1 = isEqual(a, b);
    int result2 = isEqual(a, c1); // 修改变量名以匹配新的声明
    int result3 = isEqual(b, d);
    int result4 = isEqual(d, d);
    printf("%d\n", result1); // Expected: 0
    printf("%d\n", result2); // Expected: 1
    printf("%d\n", result3); // Expected: 0
    printf("%d\n", result4); // Expected: 1
    return 0;
}
