int a;
int b;
int result;

int compare(int x, int y) {
    int diff;
    int sq;
    diff = x - y;
    sq = diff * diff;
    /*If x == y, then sq == 0, return 1; else return 0*/
    return 1 - (sq / (sq + 1));
}

int main() {
    a = getint();
    b = getint();
    result = compare(a, b);
    printf("Result of comparison: %d\n", result);
    return 0;
}
