int a = 10;
int b = 20;
const int c = 30;
int arr[3] = {0, 0, 0}; //Declare the array as a global variable

int add(int x, int y) {
    return x + y;
}

void printSum(int arr[], int size) { //Change the parameter type to int arr[]
    int sum = 0;
    int j = 0; //Rename the variable to avoid duplicate definition
    for (j = 0; j < size; j = j + 1) {
        sum = sum + arr[j];
    }
    printf("Sum: %d\n", sum);
}

int main(){
    arr[0] = a; //Initialize the global array
    arr[1] = b;
    arr[2] = c;
    
    int result = add(a, b);
    printf("Result: %d\n", result);
    printSum(arr, 3); //Use the global array as a parameter
    int input = getint();
    printf("Input: %d\n", input);
    return 0;
}
