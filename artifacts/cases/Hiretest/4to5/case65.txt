int a;
const int b = 10;
int arr[5] = {1, 2, 3, 4, 5};

int sum(int x, int y) {
    return x + y;
}

void printArray(int arr[], int size) {
    int i;
    /*Use for instead of while, which is allowed in the statement set*/
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main() {
    int result;
    a = getint();
    result = sum(a, b);
    /*Remove the disallowed character ':', while preserving the functional meaning*/
    printf("Result %d\n", result);
    printf("Array elements\n");
    printArray(arr, 5);
    return 0;
}
