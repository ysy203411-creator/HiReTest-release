int a;
const int b = 10;
int arr[5] = {1, 2, 3, 4, 5};

int sum(int x, int y) {
    return x + y;
}

void printArray(int arr[], int size) {
    int i;
    /* 使用 for 代替 while，符合允许的语句集合 */
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main() {
    int result;
    a = getint();
    result = sum(a, b);
    /* 去掉不允许的字符 ':'，保留功能含义 */
    printf("Result %d\n", result);
    printf("Array elements\n");
    printArray(arr, 5);
    return 0;
}
