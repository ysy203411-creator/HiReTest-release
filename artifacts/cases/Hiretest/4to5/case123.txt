const int SIZE = 5;
int array[SIZE] = {1, 2, 3, 4, 5};
int getElementValue(int index) {
    return array[index];
    return 0; //Add unreachable return 0; to satisfy constraints
}
int main(){
    int index = getint();
    int value = getElementValue(index);
    printf("Value at index %d is %d\n", index, value);
    return 0;
}
