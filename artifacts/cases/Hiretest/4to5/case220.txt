const int MAX_SIZE = 10;
int globalArray[MAX_SIZE];

int sum(int a, int b) {
    return a + b;
}

void printArray(int arr[], int size) {
    int i = 0;
    while (i < size) {
        printf("%d ", arr[i]);
        i = i + 1;
    }
    printf("\n");
}

int main() {
    int localArray[MAX_SIZE];
    int total = 0;
    int i = 0;

    while (i < MAX_SIZE) {
        globalArray[i] = i;
        localArray[i] = getint();
        i = i + 1;
    }

    printArray(globalArray, MAX_SIZE);
    printArray(localArray, MAX_SIZE);
    total = sum(globalArray[0], localArray[0]);
    printf("Sum of first elements: %d\n", total);
    return 0;
}
