int arr[10];
int sum(int nums[]) {
    int total = 0;
    int i = 0;
    for (i = 0; i < 10; i = i + 1) {
        total = total + nums[i];
    }
    return total;
}
int main(){
    int i = 0;
    for (i = 0; i < 10; i = i + 1) {
        arr[i] = getint();
    }
    int result = sum(arr);
    printf("%d\n", result);
    return 0;
}
