const int a = 5;
const int b = 10;
int arr[3] = {1, 2, 3};

int sum(int x, int y) {
    return x + y;
}

int main() {
    int result = sum(a, b);
    printf("%d\n", result);

    int input;
    input = getint();

    /*Limit input to 0~2 to ensure array index validity*/
    if (input < 0) {
        input = 0;
    }
    if (input > 2) {
        input = 2;
    }

    printf("%d\n", arr[input]);
    return 0;
}
