const int SIZE = 5;
int arr[SIZE];

void initializeArray(int array[]) {
    int i = 0;
    while (i < SIZE) {
        array[i] = getint();
        i = i + 1;
    }
}

int main(){
    initializeArray(arr);
    int i = 0;
    while (i < SIZE) {
        printf("%d\n", arr[i]);
        i = i + 1;
    }
    return 0;
}
