const int SIZE = 5;
int array[SIZE];
void initializeArray() {
    const int initValues_0 = 1;
    const int initValues_1 = 2;
    const int initValues_2 = 3;
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        if (i < 3) {
            if (i == 0) {
                array[i] = initValues_0;
            } else if (i == 1) {
                array[i] = initValues_1;
            } else if (i == 2) {
                array[i] = initValues_2;
            }
        } else {
            array[i] = 0; // Expectation: Remaining elements should be initialized to 0
        }
    }
}
int main(){
    initializeArray();
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", array[i]); // Expected output: 1 2 3 0 0
    }
    return 0;
}
