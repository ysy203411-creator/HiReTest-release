const int size = 10;
int globalArray[5] = {0}; //Initialize global array

void initializeArray(int arr[], int length) {
    int i = 0;
    /*Use while loop instead of for*/
    while (i < length) {
        arr[i] = i * 2;
        i = i + 1;
    }
    return;
}

int main() {
    int localArray[3];
    int i = 0;

    /*Manually initialize local array elements, avoid initialization list syntax*/
    localArray[0] = 1;
    localArray[1] = 2;
    localArray[2] = 0;

    initializeArray(globalArray, 5);

    /*Reuse i without redeclaring it*/
    i = 0;
    while (i < 3) {
        printf("%d ", localArray[i]);
        i = i + 1;
    }
    printf("\n");

    i = 0;
    while (i < 5) {
        printf("%d ", globalArray[i]);
        i = i + 1;
    }
    printf("\n");

    return 0;
}
