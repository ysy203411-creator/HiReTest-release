int a;
int b;
int c;

int main() {
    a = 1;
    b = 0;
    c = 2;

    /*Simulate a && b: when both a and b are non-zero, a * b is non-zero, otherwise 0*/
    printf("a = %d, b = %d, c = %d\n", a, b, c);
    printf("a * b (simulate a && b, nonzero means true) = %d\n", a * b);
    printf("a * c (simulate a && c, nonzero means true) = %d\n", a * c);

    /*Use the ! operator to show true and false: !0 is 1, !non-zero is 0*/
    printf("! (a * b) = %d\n", ! (a * b));
    printf("! (a * c) = %d\n", ! (a * c));

    return 0;
}
