const int MAX_LENGTH = 100;
int globalArray[MAX_LENGTH];
int globalVar = 10;
void initializeArray(int arr[]) {
    int i = 0;
    for (i = 0; i < MAX_LENGTH; i = i + 1) {
        arr[i] = i * 2;
    }
}
int sumArray(int arr[]) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < MAX_LENGTH; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(globalArray);
    int totalSum = sumArray(globalArray);
    printf("Total sum of array elements: %d\n", totalSum);
    return 0;
}
