const int MAX_SIZE = 100;
int globalArray[MAX_SIZE];
void initializeArray(int array[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        array[i] = i * 2;
    }
}
int sumArray(int array[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + array[i];
    }
    return sum;
}
int main(){
    initializeArray(globalArray, MAX_SIZE);
    int totalSum = sumArray(globalArray, MAX_SIZE);
    printf("The total sum of the array is: %d\n", totalSum);
    return 0;
}
