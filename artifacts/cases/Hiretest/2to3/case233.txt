const int MAX_SIZE = 5;
int globalArray[MAX_SIZE] = {1, 2, 3, 4, 5};
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
void printSum(int arr[], int size) {
    int total = sumArray(arr, size);
    printf("Sum of array elements: %d\n", total);
}
int main(){
    printSum(globalArray, MAX_SIZE);
    return 0;
}
