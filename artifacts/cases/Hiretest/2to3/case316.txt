int arr[3] = {1, 2, 3};  //Change to variable array
int sum(int arr[], int size) {  //Change parameter type to int[]
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    int result = sum(arr, 3);  //Change parameter to variable array name
    printf("Sum of array elements: %d\n", result);
    return 0;
}
