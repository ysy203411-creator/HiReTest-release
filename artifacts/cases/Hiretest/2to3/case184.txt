const int MAX = 10;
int arr[MAX];
int add(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int x = 5;
    int y = 10;
    int result = add(x, y);
    
    printResult(result);
    
    arr[0] = result;
    arr[1] = add(arr[0], x);
    
    printf("Array Element: %d\n", arr[1]);
    
    return 0;
}
