const int MAX = 10;
int array[MAX];
int add(int a, int b) {
    return a + b;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int x = 5;
    int y = 10;
    int result = add(x, y;
    array[0] = result;
    printArray(array, MAX);
    return 0;
}
