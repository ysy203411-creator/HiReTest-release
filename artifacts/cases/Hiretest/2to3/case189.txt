const int MAX_SIZE = 10;
int globalVar = 5;
int globalArray[MAX_SIZE] = {1, 2, 3, 4, 5};
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < MAX_SIZE; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int add(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 10;
    int localArray[MAX_SIZE];
    int i = 0;
    for (i = 0; i < MAX_SIZE; i = i + 1) {
        localArray[i] = i + 1;
    }
    printf("Global Variable: %d\n", globalVar);
    printArray(globalArray);
    printf("Local Variable: %d\n", localVar);
    printArray(localArray);
    int sum = add(globalVar, localVar);
    printf("Sum: %d\n", sum);
    return 0;
}
