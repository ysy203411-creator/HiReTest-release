const int MAX = 10;
int array[MAX];
int add(int a, int b) {
    return a + b;
}
void printSum(int x, int y) {
    int sum = add(x, y);
    printf("Sum: %d\n", sum);
}
int main(){
    int a = 5;
    int b = 3;
    int result = add(a, b);
    printf("Result: %d\n", result);
    
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        array[i] = i;
    }
    
    printSum(a, b);
    
    return 0;
}
