const int CONST_VAL = 10;
int global_var = 5;
int add(int a, int b) {
    return a + b;
}
void printConst() {
    printf("Constant value: %d\n", CONST_VAL);
}
int main(){
    int local_var = 3;
    printConst();
    int result = add(global_var, local_var);
    printf("Result: %d\n", result);
    return 0;
}
