const int MAX = 10;
int numbers[MAX];
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int value) {
    printf("Sum: %d\n", value);
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        numbers[i] = i;
    }
    int result = sum(numbers, MAX);
    printSum(result);
    return 0;
}
