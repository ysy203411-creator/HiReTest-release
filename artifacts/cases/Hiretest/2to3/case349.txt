const int MAX = 10;
int arr[MAX];
void processArray(int array[]) {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        array[i] = i * i;
    }
}
int main(){
    processArray(arr);
    int sum = 0;
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        sum = sum + arr[i];
    }
    printf("Sum of squares: %d\n", sum);
    if (sum > 0) {
        printf("Positive sum\n");
    } else {
        printf("Non-positive sum\n");
    }
    return 0;
}
