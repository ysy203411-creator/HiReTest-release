const int a = 5;
int b = 10;
void printValues(int x, int y) {
    printf("Values: %d, %d\n", x, y);  //Correct printf call
}
int main(){
    int c = a + b;
    int d = 0;
    
    if (c > 10) {
        d = c - 5;  //Correct if statement
    } else {
        d = c + 5;  //Correct else statement
    }
    
    printValues(c, d);  //Correct function call
    
    return 0;
}
