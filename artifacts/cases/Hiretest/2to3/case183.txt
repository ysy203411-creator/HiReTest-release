const int MAX = 10;
int arr[MAX];
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
void initializeArray() {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * i;
    }
}
int main(){
    initializeArray();
    printArray(arr, MAX);
    return 0;
}
