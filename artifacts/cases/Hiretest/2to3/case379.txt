const int MAX_VALUE = 100;
int numbers[MAX_VALUE];
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * 2;
    }
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    initializeArray(numbers, MAX_VALUE);
    printArray(numbers, MAX_VALUE);
    return 0;
}
