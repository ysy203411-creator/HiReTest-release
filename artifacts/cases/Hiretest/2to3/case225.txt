const int MAX = 10;
int arr[MAX];
int add(int a, int b) {
    return a + b;
}
void printArray(int[] array) {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        printf("%d\n", array[i]);
    }
}
int main(){
    int x = 5;
    int y = 3;
    int result = add(x, y);
    printf("Result: %d\n", result);
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * 2;
    }
    printArray(arr);
    return 0;
}
