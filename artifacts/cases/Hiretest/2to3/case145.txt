const int CONST_VAL = 10;
int global_var = 5;
int add(int a, int b) {
    return a + b;
}
void print_result(int result) {
    printf("Result: %d\n", result);
    return;
}
int main(){
    int local_var = CONST_VAL;
    int result = add(global_var, local_var);
    print_result(result);
    return 0;
}
