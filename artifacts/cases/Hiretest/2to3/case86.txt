const int MAX = 10;
int array[MAX];
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * i;
    }
}
int calculateSum(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(array, MAX);
    int sum = calculateSum(array, MAX);
    printf("Sum of squares: %d\n", sum);
    return 0;
}
