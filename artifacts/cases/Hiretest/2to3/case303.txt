const int SIZE = 5;
int global_array[SIZE] = {1, 2, 3, 4, 5};
int sum_array(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int local_array[3] = {10, 20, 30};
    int result1 = sum_array(global_array, SIZE);
    int result2 = sum_array(local_array, 3);
    printf("Sum of global_array: %d\n", result1);
    printf("Sum of local_array: %d\n", result2);
    return 0;
}
