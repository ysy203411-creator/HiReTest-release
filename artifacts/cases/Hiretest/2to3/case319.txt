const int SIZE = 10;
int globalArray[SIZE];
int add(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int a = 5;
    int b = 3;
    int result = add(a, b);
    printResult(result);
    globalArray[0] = result;
    if (globalArray[0] > 0) {
        printf("Positive result\n");
    } else {
        printf("Non-positive result\n");
    }
    return 0;
}
