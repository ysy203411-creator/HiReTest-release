const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int arr[], int size) {
    int total = sum(arr, size);
    printf("Sum of array elements: %d\n", total);
}
int main(){
    int localArray[SIZE] = {5, 4, 3, 2, 1};
    printSum(globalArray, SIZE);
    printSum(localArray, SIZE);
    return 0;
}
