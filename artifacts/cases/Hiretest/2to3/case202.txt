const int SIZE = 5;
int arr[SIZE] = {1, 2, 3, 4, 5};
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int calculateSum(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int sum = calculateSum(arr, SIZE);
    printf("Sum of array elements: %d\n", sum);
    printArray(arr, SIZE);
    return 0;
}
