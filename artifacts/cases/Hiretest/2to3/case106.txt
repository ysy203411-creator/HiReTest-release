const int SIZE = 5;
int array[SIZE];
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int arr[], int length) {
    int total = sum(arr, length);
    printf("Sum: %d\n", total);
}
int main(){
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        array[i] = i + 1;
    }
    printSum(array, SIZE);
    return 0;
}
