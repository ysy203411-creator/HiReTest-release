const int MAX = 100;
int numbers[MAX];
int add(int a, int b) {
    return a + b;
}
void printSum(int arr[], int size) {  // 修改函数定义以符合约束
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = add(sum, arr[i]);
    }
    printf("Sum: %d\n", sum);
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        numbers[i] = i;
    }
    printSum(numbers, MAX);  // 调用时传递数组名
    return 0;
}
