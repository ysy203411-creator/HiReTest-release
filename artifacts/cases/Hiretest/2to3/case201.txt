const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
int sumArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
void printSum(int sum) {
    printf("Sum of array elements: %d\n", sum);
}
int main(){
    int localArray[SIZE];
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        localArray[i] = globalArray[i];
    }
    int totalSum = sumArray(localArray, SIZE);
    printSum(totalSum);
    return 0;
}
