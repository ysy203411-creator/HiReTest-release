const int MAX_LENGTH = 1024;
int globalArray[MAX_LENGTH];
int globalVar = 0;
int sumArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
void initializeArray(int arr[], int length) {
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        arr[i] = i + 1;
    }
}
int main(){
    initializeArray(globalArray, MAX_LENGTH);
    int totalSum = sumArray(globalArray, MAX_LENGTH);
    printf("The total sum is: %d\n", totalSum);
    return 0;
}
