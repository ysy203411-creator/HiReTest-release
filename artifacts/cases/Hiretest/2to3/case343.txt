const int MAX = 100;
int numbers[MAX];
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void print_result(int result) {
    printf("The result is: %d\n", result);
}
int main(){
    numbers[0] = 10;
    numbers[1] = 20;
    numbers[2] = 30;
    int result = sum(numbers, 3);
    print_result(result);
    if(result > 50 && result < 100) {
        printf("Result is between 50 and 100\n");
    } else {
        printf("Result is not between 50 and 100\n");
    }
    return 0;
}
