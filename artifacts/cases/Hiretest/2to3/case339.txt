const int CONST_VAL = 10;
int global_var = 5;
int arr[5] = {1, 2, 3, 4, 5};
int multiply(int a, int b) {
    return a * b;
}
void print_result(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int local_var = 3;
    int result = multiply(global_var, local_var);
    print_result(result);
    int index = 2;
    int array_result = arr[index] * CONST_VAL;
    print_result(array_result);
    return 0;
}
