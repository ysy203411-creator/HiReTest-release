const int MAX_SIZE = 10;
int globalArray[MAX_SIZE];
void initArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * i;
    }
}
int computeSum(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initArray(globalArray, MAX_SIZE);
    int totalSum = computeSum(globalArray, MAX_SIZE);
    printf("Total Sum: %d\n", totalSum);
    return 0;
}
