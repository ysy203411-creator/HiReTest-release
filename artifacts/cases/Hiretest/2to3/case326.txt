const int SIZE = 5;
int arr[SIZE] = {1, 2, 3, 4, 5};
int sum(int a, int b) {
    return a + b;
}
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int result = sum(arr[0], arr[1]);
    printf("Sum: %d\n", result);
    printArray(arr);
    return 0;
}
