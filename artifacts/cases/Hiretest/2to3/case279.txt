const int SIZE = 5;
int array[SIZE] = {1, 2, 3, 4, 5};
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printArray(int arr[], int length) {
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int result;
    printArray(array, SIZE);
    result = sum(array, SIZE);
    printf("Sum: %d\n", result);
    return 0;
}
