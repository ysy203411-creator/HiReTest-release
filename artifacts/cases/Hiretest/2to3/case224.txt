const int MAX = 10;
int arr[MAX];
int sum(int a[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + a[i];
    }
    return total;
}
void printSum(int a[], int length) {
    int total = sum(a, length);
    printf("Sum is: %d\n", total);
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i + 1;
    }
    printSum(arr, MAX);
    return 0;
}
