const int MAX = 10;
int numbers[MAX];
int sum(int arr[], int len) {
    int total = 0;
    int i = 0;
    for (i = 0; i < len; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int arr[], int len) {
    int total = sum(arr, len);
    printf("Sum is: %d\n", total);
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        numbers[i] = i + 1;
    }
    printSum(numbers, MAX);
    return 0;
}
