const int SIZE = 5;
int sum(int arr[]) {
    int total = 0;
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printMessage() {
    printf("Hello, World!\n");
    return;
}
int main(){
    int numbers[SIZE] = {1, 2, 3, 4, 5};
    int result = sum(numbers);
    printf("Sum: %d\n", result);
    printMessage();
    return 0;
}
