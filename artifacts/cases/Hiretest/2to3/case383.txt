const int MAX = 10;
int globalArray[MAX] = {1, 2, 3, 4, 5};
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
void printSum(int sum) {
    printf("Sum is: %d\n", sum);
}
int main(){
    int localArray[MAX] = {6, 7, 8, 9, 10};
    int totalSum = sumArray(globalArray, MAX) + sumArray(localArray, MAX);
    printSum(totalSum);
    return 0;
}
