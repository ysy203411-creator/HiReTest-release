const int MAX = 10;
int arr[MAX];
int sum(int a[], int n) {
    int total = 0;
    int i = 0;
    for (i = 0; i < n; i = i + 1) {
        total = total + a[i];
    }
    return total;
}
int main(){
    int i;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i;
    }
    int totalSum = sum(arr, MAX);
    printf("Total sum is: %d\n", totalSum);
    return 0;
}
