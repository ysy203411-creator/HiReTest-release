const int CONST_VALUE = 10;
int globalVar = CONST_VALUE + 5;
int globalArray[5] = {1, 2, 3};
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < 5; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int localVar = getint();
    int localArray[3];
    localArray[0] = localVar;
    localArray[1] = CONST_VALUE;
    localArray[2] = globalVar;
    printArray(globalArray);
    printArray(localArray);
    if (localVar > 0 && localVar < 100) {
        printf("Valid input\n");
    } else {
        printf("Invalid input\n");
    }
    return 0;
}
