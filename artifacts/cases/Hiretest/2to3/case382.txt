const int CONST_VAL = 10;
int globalVar = 5;
int globalArray[3] = {1, 2};
void printSum(int a, int b) {
    int sum = a + b;
    printf("Sum: %d\n", sum);
}
int calculateDifference(int x, int y) {
    return x - y;
}
int main(){
    int localVar = 3;
    globalArray[2] = 4;
    int result = calculateDifference(globalVar, localVar);
    printf("Difference: %d\n", result);
    
    if (CONST_VAL == 10 || globalVar == 5) {
        printSum(CONST_VAL, globalVar);
    }
    int i = 0;
    for (i = 0; i < 3; i = i + 1) {
        printf("Array[%d]: %d\n", i, globalArray[i]);
    }
    return 0;
}
