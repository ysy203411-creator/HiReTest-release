const int SIZE = 5;
int array[SIZE] = {1, 2, 3, 4, 5};
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int a = 10;
    int b = 20;
    int result = a + b;
    printf("Result: %d\n", result);
    printArray(array, SIZE);
    if (a > b) {
        printf("a is greater than b\n");
    } else {
        printf("a is not greater than b\n");
    }
    return 0;
}
