const int MAX = 10;
int numbers[MAX];
void initializeArray() {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        numbers[i] = i;
    }
}
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray();
    int total = sumArray(numbers, MAX);
    printf("Total sum is: %d\n", total);
    return 0;
}
