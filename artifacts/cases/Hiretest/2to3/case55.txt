const int SIZE = 10;
int globalArray[SIZE] = {1, 2, 3, 4, 5, 0, 0, 0, 0, 0}; // 补全初始化列表
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int total) {
    printf("Sum is: %d\n", total);
}
int main(){
    int localArray[SIZE] = {6, 7, 8, 9, 10, 0, 0, 0, 0, 0}; // 补全初始化列表
    int globalSum = sum(globalArray, SIZE);
    int localSum = sum(localArray, SIZE);
    printSum(globalSum);
    printSum(localSum);
    return 0;
}
