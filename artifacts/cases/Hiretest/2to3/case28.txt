const int MAX = 10;
int globalArray[MAX];
void initializeArray(int arr[]) {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * 2;
    }
}
int processArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(globalArray);
    int result = processArray(globalArray, MAX);
    printf("Sum of array elements: %d\n", result);
    return 0;
}
