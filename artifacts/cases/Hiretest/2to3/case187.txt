const int a = 5;
int b = 10;
int arr[5] = {1, 2, 3, 4, 5};
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int sum(int x, int y) {
    return x + y;
}
int main(){
    int result = sum(a, b);
    printf("Sum: %d\n", result);
    printArray(arr, 5);
    int c = 15;
    if (c > 10) {
        printf("c is greater than 10\n");
    } else {
        printf("c is not greater than 10\n");
    }
    return 0;
}
