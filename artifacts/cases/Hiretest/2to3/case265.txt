const int MAX = 100;
int values[MAX] = {1, 2, 3, 4, 5}; //Change to global array
int array[MAX];
int sum(int[] arr, int n) {
    int total = 0;
    int i = 0;
    for (i = 0; i < n; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int result) {
    printf("Sum is: %d\n", result);
}
int main(){
    int n = 5; //Use the length of the global array
    int result = sum(values, n); //Pass the global array to the sum function
    printSum(result);
    return 0;
}
