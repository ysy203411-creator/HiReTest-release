const int CONST_VAL = 10;
int global_var = 5;
int add(int a, int b) {
    return a + b;
}
void print_sum(int x, int y) {
    int sum = add(x, y);
    printf("Sum: %d\n", sum);
}
int main(){
    int local_var = 3;
    int result = add(global_var, local_var);
    print_sum(result, CONST_VAL);
    int arr[5] = {1, 2, 3, 4, 5};
    printf("Array element: %d\n", arr[2]);
    if (result > 0) {
        printf("Result is positive.\n");
    } else {
        printf("Result is non-positive.\n");
    }
    return 0;
}
