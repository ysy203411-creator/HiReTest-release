const int a = 5;
int b = 10;
int array[5] = {1, 2, 3, 4, 5};
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < 5; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

    printf("Sum: %d\n", x);
    // This is a single line comment

    */
