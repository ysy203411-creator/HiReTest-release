const int MAX = 10;
int array[MAX];
void process(int arr[]) {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * i;
    }
}
int main(){
    int x = 5;
    int y = 10;
    int result;
    
    if (x < y & x > 0) { // a类错误：使用单独的&
        result = x + y;
    } else {
        result = x - y;
    }
    
    printf("Result: %d\n", result);
    process(array);
    
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        printf("array[%d] = %d\n", i, array[i]);
    }
    
    return 0;
}
