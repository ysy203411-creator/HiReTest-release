const int MAX = 10;
int globalArray[MAX] = {1, 2, 3, 4, 5};
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int localArray[5] = {6, 7, 8, 9, 10};
    int result1 = sumArray(globalArray, MAX);
    int result2 = sumArray(localArray, 5);
    printf("Sum of globalArray: %d\n", result1);
    printf("Sum of localArray: %d\n", result2);
    return 0;
}
