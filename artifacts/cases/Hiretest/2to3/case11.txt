const int CONST_VAL = 10;
const int SIZE = 5;
int globalVar = 5;
int array[5] = {1, 2, 3, 0, 0}; // 补全初始化列表
int add(int a, int b) {
    return a + b;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int localVar = 3;
    int result = add(globalVar, localVar);
    printf("Result: %d\n", result);
    printf("Constant Value: %d\n", CONST_VAL);
    printArray(array, SIZE);
    int sortedArray[SIZE] = {5, 3, 4, 1, 2}; // 补全初始化列表
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        int j = i + 1;
        for (j = i + 1; j < SIZE; j = j + 1) {
            if (sortedArray[i] > sortedArray[j]) {
                int temp = sortedArray[i];
                sortedArray[i] = sortedArray[j];
                sortedArray[j] = temp;
            }
        }
    }
    printf("Sorted Array: ");
    printArray(sortedArray, SIZE);
    return 0;
}
