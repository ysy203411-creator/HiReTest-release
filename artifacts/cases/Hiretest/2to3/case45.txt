const int CONST_VAL = 10;
int globalVar = 5;
int add(int a, int b) {
    return a + b;
}
int multiply(int a, int b) {
    return a * b;
}
int main(){
    int result;
    //Test function call, no parameters
    result = add(CONST_VAL, globalVar);
    printf("Result of add: %d\n", result);
    
    //Test function call, with parameters
    result = multiply(globalVar, CONST_VAL);
    printf("Result of multiply: %d\n", result);
    
    //Test function call, missing closing parenthesis error
    result = add(globalVar, CONST_VAL;
    printf("This line should not compile due to missing ')'\n");
    
    return 0;
}
