const int a = 10;
const int b = 20;
int arr[5] = {1, 2, 3, 4, 5};
int compare(int x, int y) {
    return x == y;
}
int main(){
    int result = compare(a, b);
    if (result & arr[0]) {  // Use of a separate & should be identified as an error
        printf("Equal\n");
    } else {
        printf("Not equal\n");
    }
    return 0;
}
