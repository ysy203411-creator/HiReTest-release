const int SIZE = 10;
int arr[SIZE];
int add(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result: %d\n", result);
}
int main(){
    int x = 5;
    int y = 10;
    int sum = add(x, y);
    printResult(sum);
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        arr[i] = i * 2;
    }
    int z = arr[5];
    printf("Array element: %d\n", z);
    if (x < y) {
        printf("x is less than y\n");
    }
    return 0;
}
