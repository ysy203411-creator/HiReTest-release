const int SIZE = 5;
int arr[SIZE] = {1, 2, 3, 4, 5};
const int constArr[SIZE] = {1, 2, 3, 4, 5};
int main(){
    int index = 2;
    arr[index] = 10; //Modify array elements
    printf("arr[%d] = %d\n", index, arr[index]); //Print array elements
    index = 3;
    //constArr[index] = 10; // This line should cause a compilation error because constArr is a constant array and cannot be modified
    printf("constArr[%d] = %d\n", index, constArr[index]); //Print constant array elements
    return 0;
}
