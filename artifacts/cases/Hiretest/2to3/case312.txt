const int CONST_VAL = 5;
int globalVar = 10;
int add(int a, int b) {
    return a + b;
}
void printSum(int x, int y) {
    int sum = add(x, y);
    printf("Sum: %d\n", sum);
}
int main(){
    int localVar = 20;
    int result = add(globalVar, localVar);
    printf("Result: %d\n", result);
    printSum(CONST_VAL, result);
    return 0;
}
