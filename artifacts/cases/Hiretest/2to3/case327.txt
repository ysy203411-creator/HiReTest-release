const int MAX = 10;
int arr[MAX];
int compute(int x) {
    return x * 2;
}
void process(int data[]) {
    int j = 0;  // 修改局部变量名以避免重名
    for (j = 0; j < MAX; j = j + 1) {
        data[j] = compute(data[j]);
    }
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i;
    }
    
    process(arr);  // 实参为数组名，符合约束
    int result = 0;
    int m = 0;  // 修改局部变量名以避免重名
    for (m = 0; m < MAX; m = m + 1) {
        result = result + arr[m];
    }
    printf("Result: %d\n", result);
    return 0;
}
