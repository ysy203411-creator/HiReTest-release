const int CONST_VALUE = 10;
int globalVar = 5;
int array[5] = {1, 2, 3, 0, 0}; // 修改初始化列表，补全未初始化元素
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < 5; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int add(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 3;
    static int staticVar;
    printf("Const Value: %d\n", CONST_VALUE);
    printf("Global Var: %d\n", globalVar);
    printf("Local Var: %d\n", localVar);
    printf("Static Var: %d\n", staticVar);
    
    int result = add(globalVar, localVar);
    printf("Add Result: %d\n", result);
    
    printArray(array);
    
    return 0;
}
