const int CONST_VAL = 10;
int globalVar = 5;
void testFunction(int param) {
    int localVar = param + CONST_VAL;
    if (localVar > 10) {
        printf("Value is greater than 10\n");
    } else {
        printf("Value is 10 or less\n");
    }
}
int main(){
    int array[5] = {1, 2, 3, 4, 5};
    int i = 0;
    for (i = 0; i < 5; i = i + 1) {
        array[i] = array[i] * 2;
        if (array[i] > 5) {
            printf("Element %d is greater than 5\n", i);
        }
    }
    testFunction(globalVar);
    return 0;
}
