const int SIZE = 10;
int numbers[SIZE];
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i;
    }
    return total;
}
int main(){
    numbers[0] = 1;
    numbers[1] = 2;
    numbers[2] = 3;
    numbers[3] = 4;
    numbers[4] = 5;
    numbers[5] = 6;
    numbers[6] = 7;
    numbers[7] = 8;
    numbers[8] = 9;
    numbers[9] = 10;
    int result = sum(numbers, SIZE);
    printf("Sum: %d\n", result);
    return 0;
}
