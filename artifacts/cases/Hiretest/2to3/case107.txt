const int MAX = 10;
int numbers[MAX] = {0}; // 初始化全局变量数组
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void initializeArray() {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        numbers[i] = i + 1;
    }
}
int main(){
    initializeArray();
    int totalSum = sum(numbers, MAX);
    printf("Total sum is: %d\n", totalSum);
    return 0;
}
