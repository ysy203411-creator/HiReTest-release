const int SIZE = 5;
int array[SIZE] = {1, 2, 3, 4}; // 初始化缺少一个元素，测试大括号处理
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int total) {
    printf("Sum: %d\n", total);
}
int main(){
    int result = sum(array, SIZE);
    printSum(result);
    return 0;
}
