const int SIZE = 10;
int globalArray[10];  // 修改为编译期可求值的非负整数常量表达式
int sumArray(int arr[], int len) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < len; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
void printSum(int arr[], int len) {
    int sum = sumArray(arr, len);
    printf("Sum of array elements: %d\n", sum);
}
int main(){
    int localArray[5] = {1, 2, 3, 4, 5};
    printSum(localArray, 5);
    return 0;
}
