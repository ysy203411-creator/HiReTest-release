const int MAX = 100;
int globalVar = 10;
int array[MAX];
int sum(int a, int b) {
    return a + b;
}
void printSum(int a, int b) {
    int result = sum(a, b);
    printf("Sum: %d\n", result);
    return;
}
int main(){
    int localVar = 5;
    int anotherVar = 15;
    int result = sum(localVar, anotherVar);
    printf("Result: %d\n", result);
    int i = 0;
    for (i = 0; i < 10; i = i + 1) {
        array[i] = i * i;
        printf("Array[%d]: %d\n", i, array[i]);
    }
    printSum(globalVar, localVar);
    return 0;
}
