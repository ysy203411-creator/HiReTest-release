const int MAX = 10;
int arr[MAX];
int value = 5;
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * i;
    }
}
int main(){
    initializeArray(arr, MAX);
    int sum = 0;
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        sum = sum + arr[i];
    }
    printf("Sum of squares: %d\n", sum);
    return 0;
}
