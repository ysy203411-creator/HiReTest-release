const int SIZE = 5;
int global_array[SIZE] = {1, 2, 3, 4, 5};
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void print_result(int result) {
    printf("The sum is: %d\n", result);
}
int main(){
    int local_array[SIZE] = {5, 4, 3, 2, 1};
    int result1 = sum(global_array, SIZE);
    int result2 = sum(local_array, SIZE);
    print_result(result1);
    print_result(result2);
    return 0;
}
