const int SIZE = 5;
const int INIT_VALUES[SIZE] = {1, 2, 3, 4, 5};
int globalArray[SIZE] = {INIT_VALUES[0], INIT_VALUES[1], INIT_VALUES[2], INIT_VALUES[3], INIT_VALUES[4]};
void processArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int localArray[SIZE];
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        localArray[i] = globalArray[i] * 2;
    }
    
    processArray(localArray, SIZE);
    
    return 0;
}
