const int CONST_VAL = 10;
int global_var = 5;
int array_var[5] = {1, 2, 3, 4, 5};
void printValue(int value) {
    printf("Value: %d\n", value);
    return;
}
int calculateSum(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int local_var = 3;
    int result = calculateSum(array_var, 5);
    printValue(result);
    if (local_var > 2) {
        printf("Local variable is greater than 2\n");
    }
    return 0;
}
