const int MAX = 10;
int globalVar = 5;
int array[MAX];
void process(int a, int b) {
    int sum = a + b;
    printf("Sum: %d\n", sum);
}
int compute(int x) {
    return x * x;
}
int main(){
    int localVar = 3;
    int result = compute(localVar);
    printf("Result: %d\n", result);
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        array[i] = i * 2;
    }
    process(globalVar, array[2]);
    printf("Array element: %d\n", array[5]);
    return 0;
}
