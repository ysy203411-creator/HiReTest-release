const int MAX = 10;
int globalVar = 5;
int numbers[MAX];
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * 2;
    }
}
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(numbers, MAX);
    int total = sumArray(numbers, MAX);
    printf("The sum of the array is: %d\n", total);
    return 0;
}
