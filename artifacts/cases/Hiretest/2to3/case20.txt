const int SIZE = 5;
int numbers[SIZE] = {1, 2, 3, 4, 5};
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int arr[], int length) {
    int result = sum(arr, length);
    printf("The sum is: %d\n", result);
}
