const int MAX = 100;
int globalVar = 10;
int array[MAX];
void printMessage() {
    printf("Hello, World!\n");
}
int add(int a, int b) {
    return a + b;
}
int main(){
    int localVar = 5;
    localVar = globalVar + 2;
    array[0] = localVar;
    
    if (localVar > 0) {
        printMessage();
    } else {
        printf("localVar is non-positive\n");
    }
    int result = add(localVar, array[0]);
    printf("Result: %d\n", result);
    return 0;
}
