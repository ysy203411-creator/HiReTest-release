const int MAX = 10;
int arr[MAX];
int sum(int a, int b) {
    return a + b;
}
void printSum(int a, int b) {
    printf("Sum is: %d\n", sum(a, b));
}
int main(){
    int x = 5;
    int y = 7;
    printSum(x, y);
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * i;
    }
    printf("Square of 3 is: %d\n", arr[3]);
    return 0;
}
