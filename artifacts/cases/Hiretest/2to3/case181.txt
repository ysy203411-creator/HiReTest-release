const int SIZE = 10;
int globalArray[SIZE];
void initializeArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        arr[i] = i * 2;
    }
}
int sumArray(int arr[]) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(globalArray);
    int total = sumArray(globalArray);
    printf("Total sum is: %d\n", total);
    return 0;
}
