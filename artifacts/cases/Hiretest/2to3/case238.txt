const int CONST_VALUE = 10;
int globalVar = 5;
int globalArray[5] = {1, 2, 3, 4, 5};
void printConst() {
    printf("Const Value: %d\n", CONST_VALUE);
}
int sumArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int localVar = 3;
    int localArray[3] = {6, 7, 8};
    printConst();
    int totalSum = sumArray(globalArray, 5) + sumArray(localArray, 3);
    printf("Total Sum: %d\n", totalSum);
    return 0;
}
