const int CONST_VAL = 10;
int globalVar = 5;
int add(int a, int b) {
    return a + b;
}
void printSum(int a, int b) {
    int sum = add(a, b);
    printf("Sum: %d\n", sum);
}
int main(){
    int localVar = 3;
    int result = add(globalVar, localVar);
    printf("Result: %d\n", result);
    int array[5] = {1, 2, 3, 4, 5};
    int i = 0;
    for (i = 0; i < 5; i = i + 1) {
        printf("Array[%d]: %d\n", i, array[i]);
    }
    printSum(CONST_VAL, result);
    return 0;
}
