const int ARRAY_SIZE = 5;
int globalArray[ARRAY_SIZE] = {1, 2, 3, 4, 5};
int globalVar = 10;
int add(int a, int b) {
    return a + b;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
    return;
}
int main(){
    int localVar = 20;
    int result = add(globalVar, localVar);
    printf("Result: %d\n", result);
    printArray(globalArray, ARRAY_SIZE);
    return 0;
}
