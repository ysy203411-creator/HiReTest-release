const int MAX_LENGTH = 10;
int globalArray[MAX_LENGTH] = {1, 2, 3, 4, 5};
int sumArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
void printSum(int sum) {
    printf("Sum of array elements: %d\n", sum);
}
int main(){
    int localArray[MAX_LENGTH];
    int i = 0;
    for (i = 0; i < MAX_LENGTH; i = i + 1) {
        localArray[i] = i + 1;
    }
    int totalSum = sumArray(localArray, MAX_LENGTH);
    printSum(totalSum);
    return 0;
}
