const int MAX_LENGTH = 100;
int globalArray[MAX_LENGTH];
int sumArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int localArray[5] = {1, 2, 3, 4, 5};
    int result = sumArray(localArray, 5);
    printf("Sum of localArray: %d\n", result);
    // Test reading to line end
    int i = 0;
    for (i = 0; i < MAX_LENGTH; i = i + 1) {
        globalArray[i] = i;
    }
    result = sumArray(globalArray, MAX_LENGTH);
    printf("Sum of globalArray: %d\n", result);
    return 0;
}
