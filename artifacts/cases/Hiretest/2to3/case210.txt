int func(int x) {
    int y = x + 1;
    return y;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0 < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int arr[3] = {10, 20, 30};
    printArray(arr, 3);
    b = func(5);
    printf("%d\n", b);
    return 0;
}
