const int NUM = 10;
int arr[NUM];
void fillArray(int a[]) {
    int i = 0;
    for (i = 0; i < NUM; i = i + 1) {
        a[i] = i * i;
    }
}
int main(){
    fillArray(arr);
    printf("Array values: ");
    int i = 0;
    for (i = 0; i < NUM; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    return 0;
}
