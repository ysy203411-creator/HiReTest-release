const int a = 10;
int b = 20;
int add(int x, int y) {
    return x + y;
}
void printSum(int[] arr) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < 5; i = i + 1) {
        sum = sum + arr[i];
    }
    printf("Sum: %d\n", sum);
}
int main(){
    int numbers[5] = {1, 2, 3, 4, 5};
    int result = add(a, b);
    printf("Result: %d\n", result);
    printSum(numbers);
    return 0;
}
