const int SIZE = 10;
int globalArray[SIZE];
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    int localArray[SIZE];
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        localArray[i] = i * 2;
    }
    printf("Sum of globalArray: %d\n", sum(globalArray, SIZE));
    printf("Sum of localArray: %d\n", sum(localArray, SIZE));
    return 0;
}
