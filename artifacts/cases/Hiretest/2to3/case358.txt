const int MAX = 10;
int globalArray[MAX];
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int arr[], int length) {
    int result = sum(arr, length);
    printf("Sum: %d\n", result);
    return;
}
int main(){
    int localArray[MAX];
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        localArray[i] = i + 1;
    }
    printSum(localArray, MAX);
    return 0;
}
