const int a = 5;
int b;
int c[3] = {1, 2, 3};
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int sum(int x, int y) {
    return x + y;
}
int main(){
    int result = sum(a, b);
    printf("Sum: %d\n", result);
    printArray(c, 3);
    return 0;
}
