const int SIZE = 5;
int arr[SIZE] = {1, 2, 3, 4, 5}; // 修改为变量数组
int sumArray(int a[], int size) { // 修改参数类型
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + a[i];
    }
    return sum;
}
int main(){
    const int x = 10;
    const int y = x + 20;
    int result = sumArray(arr, SIZE);
    printf("Sum of array: %d\n", result);
    printf("Value of y: %d\n", y);
    return 0;
}
