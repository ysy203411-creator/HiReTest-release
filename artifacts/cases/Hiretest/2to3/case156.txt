const int MAX_LENGTH = 100;
int testArray[MAX_LENGTH];
void initializeArray(int arr[], int length) {
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        arr[i] = i * 2;
    }
}
int calculateSum(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    const int LENGTH = 10;
    int localArray[LENGTH];
    initializeArray(localArray, LENGTH);
    int totalSum = calculateSum(localArray, LENGTH);
    printf("Total sum: %d\n", totalSum);
    return 0;
}
