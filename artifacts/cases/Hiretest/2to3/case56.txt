const int LENGTH = 5;
int numbers[LENGTH] = {1, 2, 3, 4, 5};
int sum(int array[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + array[i];
    }
    return total;
}
void printSum(int value) {
    printf("Sum: %d\n", value);
}
int main(){
    int result = sum(numbers, LENGTH);
    printSum(result);
    return 0;
}
