const int MAX = 10;
int array[MAX];
int sum(int arr[], int size) {
    int total = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printSum(int arr[], int size) {
    int total = sum(arr, size);
    printf("Sum: %d\n", total);
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        array[i] = i + 1;
    }
    printSum(array, MAX);
    return 0;
}
