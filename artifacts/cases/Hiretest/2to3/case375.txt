const int MAX = 100;
int globalArray[MAX];
int add(int a, int b) {
    return a + b;
}
void printSum(int x, int y) {
    int sum = add(x, y);
    printf("Sum: %d\n", sum);
}
int main(){
    int a = 5;
    int b = 10;
    printSum(a, b);
    int i = 0;
    for (i = 0; i < 10; i = i + 1) {
        globalArray[i] = i * i;
        printf("globalArray[%d] = %d\n", i, globalArray[i]);
    }
    return 0;
}
