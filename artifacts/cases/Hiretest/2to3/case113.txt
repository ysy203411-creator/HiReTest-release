const int MAX = 10;
int array[MAX];
int sum(int a, int b) {
    return a + b;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int a = 5;
    int b = 10;
    int result = sum(a, b);
    printf("Sum: %d\n", result);
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        array[i] = i * i;
    }
    printArray(array, MAX);
    return 0;
}
