const int MAX = 10;
const int globalArray[MAX] = {1, 2, 3}; // 修改为常量数组
int sum(int a, int b) {
    return a + b;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int localArray[5] = {4, 5, 6, 7, 8};
    int result = sum(globalArray[0], localArray[0]);
    printf("Sum: %d\n", result);
    printArray(globalArray, MAX);
    printArray(localArray, 5);
    return 0;
}
