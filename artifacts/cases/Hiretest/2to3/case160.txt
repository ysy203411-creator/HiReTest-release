const int MAX = 10;
int globalArray[MAX];
int add(int a, int b) {
    return a + b;
}
void printResult(int result) {
    printf("Result is: %d\n", result);
}
int main(){
    int x = 5;
    int y = 3;
    int result = add(x, y);
    if (result > 0) {
        printResult(result;
    } else {
        printf("No positive result\n");
    }
    globalArray[0] = result;
    printf("Array first element: %d\n", globalArray[0]);
    return 0;
}
