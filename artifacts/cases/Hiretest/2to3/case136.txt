const int MAX = 10;
int sum(int arr[], int n) {
    int total = 0;
    int i = 0;
    for (i = 0; i < n; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
int main(){
    int numbers[MAX];
    int idx = 0; // 修改变量名以避免重名
    for (idx = 0; idx < MAX; idx = idx + 1) {
        numbers[idx] = idx;
    }
    int result = sum(numbers, MAX);
    printf("Sum: %d\n", result);
    for (idx = 0; idx < MAX; idx = idx + 1) {
        int j = 0;
        for (j = 0; j < MAX; j = j + 1) {
            numbers[j] = idx * j;
        }
    }
    return 0;
}
