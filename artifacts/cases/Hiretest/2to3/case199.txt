const int SIZE = 5;
int global_array[SIZE] = {1, 2, 3, 4, 5};
int global_var = 10;
int sum(int a, int b) {
    return a + b;
}
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int local_var = sum(global_var, 20);
    printf("Sum: %d\n", local_var);
    printArray(global_array);
    if (local_var > 25) {
        printf("Local var is greater than 25\n");
    } else {
        printf("Local var is less than or equal to 25\n");
    }
    return 0;
}
