const int MAX = 100;
int numbers[MAX];
int current_number = 0;
int process_number(int num) {
    return num * 2;
}
void print_numbers(int[] nums, int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", nums[i]);
    }
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        numbers[i] = process_number(i);
    }
    print_numbers(numbers, MAX);
    return 0;
}
