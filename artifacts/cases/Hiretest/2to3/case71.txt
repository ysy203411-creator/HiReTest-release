const int MAX_LEVEL = 5;
int globalVar = 10;
int arr[MAX_LEVEL] = {1, 2, 3, 4, 5};
int sum(int a, int b) {
    return a + b;
}
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < MAX_LEVEL; i = i + 1) {
        printf("%d ", arr[i]);
    }
    return;
}
int main(){
    int localVar = 5;
    int result = sum(globalVar, localVar);
    printf("Sum: %d\n", result);
    printArray(arr);
    if (result > 0) {
        printf("Result is positive\n");
    } else {
        printf("Result is non-positive\n");
    }
    return 0;
}
