const int MAX = 10;
int numbers[MAX];
void initializeArray(int arr[]) {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * i;
    }
}
int sumArray(int arr[]) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(numbers);
    int totalSum = sumArray(numbers);
    printf("Total sum: %d\n", totalSum);
    return 0;
}
