const int MAX = 10;
int globalArray[MAX];
int add(int a, int b) {
    return a + b;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int localArray[MAX];
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        globalArray[i] = i;
        localArray[i] = MAX - i;
    }
    
    printf("Sum of 3 and 4 is: %d\n", add(3, 4));
    printf("Global array: ");
    printArray(globalArray, MAX);
    printf("Local array: ");
    printArray(localArray, MAX);
    
    return 0;
}
