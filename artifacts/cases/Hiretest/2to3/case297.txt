const int a[5] = {1, 2, 3, 4, 5};
int b = 10;  // 修改为变量
int c[3] = {1, 2, 3};
int add(int x, int y) {
    return x + y;
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int result = add(b, 3);  // 修改为常量表达式
    printf("Result: %d\n", result);
    printArray(c, 3);
    return 0;
}
