const int rq1_case_id_27 = 27;
const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
void assignValue(int arr[]) {  // 修改形参定义
    arr[2] = 10;
}
int main(){
    int localArray[SIZE];
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        localArray[i] = globalArray[i];
    }
    
    assignValue(localArray);  // 传入变量数组名
    
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", localArray[i]);
    }
    
    return 0;
}
