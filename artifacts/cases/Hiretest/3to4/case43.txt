const int MAX_SIZE = 10;
int globalArray[MAX_SIZE];
int anotherArray[MAX_SIZE];
void setArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * 2;
    }
}
int main(){
    int localArray[MAX_SIZE];
    setArray(globalArray, MAX_SIZE);
    setArray(localArray, MAX_SIZE);
    
    printf("Global Array: ");
    int i = 0;
    for (i = 0; i < MAX_SIZE; i = i + 1) {
        printf("%d ", globalArray[i]);
    }

    for (i = 0; i < MAX_SIZE; i = i + 1) {
        printf("%d ", localArray[i]);
    }

}
