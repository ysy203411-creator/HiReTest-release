const int rq1_case_id_21 = 21;
const int N = 10;
int arr[N];
void fillArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * i;
    }
}
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    fillArray(arr, N);
    printArray(arr, N);
    return 0;
}
