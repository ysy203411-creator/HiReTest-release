const int rq1_case_id_41 = 41;
const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
void processArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d\n", arr[i]);
    }
    return;
}
int main(){
    int localArray[SIZE];
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        localArray[i] = globalArray[i] * 2;
    }
    processArray(localArray);
    return 0;
}
