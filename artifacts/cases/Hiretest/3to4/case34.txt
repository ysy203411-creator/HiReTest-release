const int rq1_case_id_34 = 34;
const int MAX = 10;
int arr[MAX];
int sum(int a[], int n) {
    int total = 0;
    int i = 0;
    for (i = 0; i < n; i = i + 1) {
        total = total + a[i];
    }
    return total;
}
int main(){
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i + 1;
    }
    int result = sum(arr, MAX);
    printf("Sum of array elements: %d\n", result);
    return 0;
}
