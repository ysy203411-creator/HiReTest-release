const int rq1_case_id_48 = 48;
const int SIZE = 10;
int globalArray[SIZE];
void processArray(int arr[], int length) {
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        arr[i] = i * 2;
    }
}
int main(){
    int localArray[SIZE];
    processArray(localArray, SIZE);
    
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("localArray[%d] = %d\n", i, localArray[i]);
    }
    processArray(globalArray, SIZE);
    
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("globalArray[%d] = %d\n", i, globalArray[i]);
    }
    return 0;
}
