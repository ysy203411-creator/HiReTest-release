int add(int a, int b) {

}

    printf("Result: %d\n", result);
}

    int array[3] = {1, 2, 3};
    int result = add(globalVar, localVar);
    printResult(result);

        printf("Array first element equals CONST_VAL\n");
    } else {
        printf("Array first element does not equal CONST_VAL\n");
