const int rq1_case_id_59 = 59;
const int MAX_SIZE = 10;
int globalArray[MAX_SIZE] = {1, 2, 3, 4, 5};
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int localArray[MAX_SIZE] = {6, 7, 8, 9, 10};
    printArray(globalArray, MAX_SIZE);
    printArray(localArray, MAX_SIZE);
    int totalSum = sumArray(globalArray, MAX_SIZE) + sumArray(localArray, MAX_SIZE);
    printf("Total Sum: %d\n", totalSum);
    return 0;
}
