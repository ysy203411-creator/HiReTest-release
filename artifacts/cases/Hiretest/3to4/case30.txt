const int rq1_case_id_30 = 30;
const int MAX = 10;
int arr[MAX];
void processArray(int arr[]) {
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        arr[i] = i * i;
    }
}
int main(){
    processArray(arr);
    int sum = 0;
    int i = 0;
    for (i = 0; i < MAX; i = i + 1) {
        sum = sum + arr[i];
    }
    printf("Sum of squares: %d\n", sum);
    return 0;
}
