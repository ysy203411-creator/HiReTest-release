const int rq1_case_id_68 = 68;
const int SIZE = 10;
int globalArray[SIZE];
void testFunction() {
    printf("This is a test function.\n");
    return;
}
int main(){
    int localArray[SIZE];
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        globalArray[i] = i;
        localArray[i] = SIZE - i;
    }
    testFunction();
    int sum = 0;
    // Remove the redeclaration of i
    for (i = 0; i < SIZE; i = i + 1) {
        sum = sum + globalArray[i] + localArray[i];
    }
    printf("Sum: %d\n", sum);
    return 0;
}
