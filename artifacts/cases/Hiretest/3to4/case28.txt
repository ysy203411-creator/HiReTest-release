const int rq1_case_id_28 = 28;
const int SIZE = 5;
int array[SIZE] = {1, 2, 3, 4, 5};
int sum(int arr[], int length) {
    int total = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        total = total + arr[i];
    }
    return total;
}
void printResult(int result) {
    printf("Sum is: %d\n", result);
    return;
}
int main(){
    int result = sum(array, SIZE);
    printResult(result);
    return 0;
}
