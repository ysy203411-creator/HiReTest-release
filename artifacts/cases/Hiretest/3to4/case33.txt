const int rq1_case_id_33 = 33;
const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d\n", arr[i]);
    }
}
int main(){
    int localArray[SIZE];
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        localArray[i] = globalArray[i] * 2;
    }
    printArray(localArray);
    return 0;
}
