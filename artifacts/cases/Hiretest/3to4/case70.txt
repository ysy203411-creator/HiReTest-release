const int rq1_case_id_70 = 70;
const int N = 10;
int values[N];
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * 2;
    }
}
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(values, N);
    int total = sumArray(values, N);
    printf("Total sum is: %d\n", total);
    return 0;
}
