const int SIZE = 5;
int globalArray[SIZE] = {1, 2, 3, 4, 5};
int sumArray(int arr[], int length) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < length; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    int localVar = 10;
    int localArray[SIZE] = {5, 4, 3, 2, 1};
    
    int result1 = sumArray(globalArray, SIZE);
    int result2 = sumArray(localArray, SIZE);
    
    printf("Sum of globalArray: %d\n", result1);
    printf("Sum of localArray: %d\n", result2);
    
    return 0;
}
