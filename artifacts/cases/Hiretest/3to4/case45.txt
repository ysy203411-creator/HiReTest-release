const int SIZE = 10;
int arr[SIZE];
void printArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        arr[i] = i * 2;
    }
    printArray(arr);
    return 0;
}
