const int rq1_case_id_93 = 93;
const int MAX_SIZE = 10;
int globalArray[MAX_SIZE];
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * 2;
    }
}
int sumArray(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(globalArray, MAX_SIZE);
    int totalSum = sumArray(globalArray, MAX_SIZE);
    printf("Total sum: %d\n", totalSum);
    return 0;
}
