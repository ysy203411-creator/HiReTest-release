const int rq1_case_id_37 = 37;
const int SIZE = 10;
int globalArray[SIZE];
void initializeArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = i * 2;
    }
}
int calculateSum(int arr[], int size) {
    int sum = 0;
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        sum = sum + arr[i];
    }
    return sum;
}
int main(){
    initializeArray(globalArray, SIZE);
    int sum = calculateSum(globalArray, SIZE);
    printf("Sum of array elements: %d\n", sum);
    return 0;
}
