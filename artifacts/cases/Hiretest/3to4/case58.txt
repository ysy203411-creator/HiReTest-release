const int rq1_case_id_58 = 58;
const int N = 5;
int arr[N];
int sum(int a[]) {
    int total = 0;
    int i = 0;
    for (i = 0; i < N; i = i + 1) {
        total = total + a[i];
    }
    return total;
}
int main(){
    int i;
    for (i = 0; i < N; i = i + 1) {
        arr[i] = i * 2;
    }
    int result = sum(arr);
    printf("Sum of array elements: %d\n", result);
    return 0;
}
