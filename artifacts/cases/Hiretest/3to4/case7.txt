const int rq1_case_id_7 = 7;
const int SIZE = 5;
int numbers[SIZE] = {1, 2, 3, 4, 5};

int add(int a, int b) {
    return a + b;
}

void print_escaped_chars() {
    printf("Escaped characters: \\n, \\t\n");
    return;
}

int main() {
    int sum = add(numbers[1], numbers[3]);
    printf("Sum: %d\n", sum);
    print_escaped_chars();
    int i;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("Number %d: %d\n", i, numbers[i]);
    }
    return 0;
}
