const int size = 6;
const int str_size = 5;
int global_array[size] = {1, 2, 3, 4, 5, 6};
int non_const_array[str_size] = {7, 8, 9, 10, 11}; // 修改为非const数组
int main(){
    int local_array[size];
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        local_array[i] = global_array[i] + non_const_array[i % str_size];
    }
    printf("Local array: ");
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", local_array[i]);
    }
    printf("\n");
    printf("String: Hello\n");
    return 0;
}
