int globalVar = 10;
int array[5] = {1, 2, 3, 4, 5};

void modifyGlobalVar(int value) {
    globalVar = value;
}

int accessArrayElement(int index) {
    return array[index];
}

int main() {
    int input;
    input = getint();
    /*Use input in the main function to modify the global variable, avoiding passing function call expressions as parameters*/
    globalVar = input;
    printf("Global Variable: %d\n", globalVar);
    int element = accessArrayElement(2); //Modify: first get the array elements, then pass them to printf
    printf("Array Element at index 2: %d\n", element);
    return 0;
}
