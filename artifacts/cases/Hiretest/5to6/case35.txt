int globalVar = 123;

int truncateToI1(int value) {
    int tmp;
    tmp = value % 2;
    if (tmp != 0) {
        return 1;
    } else {
        return 0;
    }
}

int main() {
    int inputValue;
    int truncatedValue;
    inputValue = getint(); // Read an integer from standard input
    truncatedValue = truncateToI1(inputValue);
    printf("Original: %d, Truncated: %d\n", inputValue, truncatedValue);
    return 0;
}
