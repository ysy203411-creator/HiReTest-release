const int MAX = 10;
int arr[MAX];
int sum(int a[], int n) {
    int total = 0;
    int i = 0;
    for (i = 0; i < n; i = i + 1) {
        total = total + a[i];
    }
    return total;
}
int main(){
    int n;
    n = getint();
    if (n > MAX || n < 0) {
        printf("Error: Invalid array size\n");
        return 0;
    }
    int i = 0;
    for (i = 0; i < n; i = i + 1) {
        arr[i] = getint();
    }
    int result = sum(arr, n);
    printf("Sum: %d\n", result);
    return 0;
}
