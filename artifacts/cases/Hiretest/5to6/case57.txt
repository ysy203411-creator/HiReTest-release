int globalArray[5] = {1, 2, 3, 4, 5};
const int SIZE = 5;
void loadArray(int arr[]) {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int localArray[5] = {10, 20, 30, 40, 50};
    loadArray(globalArray);
    loadArray(localArray);
    return 0;
}
