const int SIZE = 5;
int globalArray[SIZE] = {255, 127, 128, 0, 1};
void testFunction(int a, int b, int c, int d, int e) {
    printf("%d %d %d %d %d\n", a, b, c, d, e);
}
int main(){
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        testFunction(globalArray[i], globalArray[(i + 1) % SIZE], globalArray[(i + 2) % SIZE], globalArray[(i + 3) % SIZE], globalArray[(i + 4) % SIZE]);
    }
    return 0;
}
