const int SIZE = 5;
const int ARRAY_SIZE = 10;
int globalArray[5] = {97, 98, 99, 100, 0}; // ASCII values for "abcd"
int main(){
    int localArray[10] = {104, 101, 108, 108, 111, 0}; // ASCII values for "hello"
    printf("Global Array: ");
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", globalArray[i]); // Print ASCII values
    }
    printf("\nLocal Array: ");
    for (i = 0; i < ARRAY_SIZE; i = i + 1) {
        printf("%d ", localArray[i]); // Print ASCII values
    }
    printf("\n");
    return 0;
}
