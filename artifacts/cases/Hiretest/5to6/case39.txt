const int SIZE = 5;
int arr[SIZE];

void updateArray(int index, int value) {
    if (index >= 0 && index < SIZE) {
        arr[index] = value;
    } else {
        printf("index out of range\n");
    }
}

int main() {
    int index = getint();
    int value = getint();
    updateArray(index, value);
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    return 0;
}
