int a[5] = {1, 2, 3, 4, 5}; // 修改为全局变量数组
const int b = 10;
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d", arr[i]);
        if (i < size - 1) {
            printf(", ");
        }
    }
    printf("\n");
}
int main(){
    const int maxSize = 100; // 定义一个常量作为数组的最大长度
    int size = getint();
    if (size > maxSize) {
        size = maxSize; // 限制数组长度不超过最大值
    }
    int arr[100]; // 使用常量表达式定义数组长度
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = getint();
    }
    printArray(arr, size);
    printArray(a, 5); // 使用变量数组作为实参
    printf("Constant b: %d\n", b);
    return 0;
}
