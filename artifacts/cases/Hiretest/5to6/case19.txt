int a[5] = {1, 2, 3, 4, 5}; //Convert to global array
const int b = 10;
void printArray(int arr[], int size) {
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d", arr[i]);
        if (i < size - 1) {
            printf(", ");
        }
    }
    printf("\n");
}
int main(){
    const int maxSize = 100; //Define a constant as the maximum array length
    int size = getint();
    if (size > maxSize) {
        size = maxSize; //Limit array length not to exceed the maximum value
    }
    int arr[100]; //Use a constant expression to define array length
    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        arr[i] = getint();
    }
    printArray(arr, size);
    printArray(a, 5); //Use a variable array as an argument
    printf("Constant b: %d\n", b);
    return 0;
}
