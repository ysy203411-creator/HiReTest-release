int globalVar = 42;
const int constArray[3] = {1, 2, 3};
void printGlobalVar() {
    printf("Global variable: %d\n", globalVar);
}
void printConstArray() {
    int i = 0;
    for (i = 0; i < 3; i = i + 1) {
        printf("Const array element %d: %d\n", i, constArray[i]);
    }
}
int main(){
    printGlobalVar();
    printConstArray();
    globalVar = getint();
    printf("Updated global variable: %d\n", globalVar);
    return 0;
}
