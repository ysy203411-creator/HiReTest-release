int arr[5] = {1, 2, 3, 4, 5};
int getElement(int arr[], int offset) {
    return arr[offset];
}
int main(){
    int index = getint(); //The getint() function can be used without explicit declaration
    if (index >= 0 && index < 5) {
        int value = getElement(arr, index);
        printf("Value at index %d: %d\n", index, value);
    } else {
        printf("Index out of bounds\n");
    }
    return 0;
}
