const int size = 5;
int globalArray[size];

int main() {
    const int c0 = 97;  // 'a'
    const int c1 = 10;  // '\n'
    const int c2 = 98;  // 'b'
    const int c3 = 0;
    const int c4 = 0;

    int i = 0;
    for (i = 0; i < size; i = i + 1) {
        if (i == 0) {
            globalArray[i] = c0;
        } else if (i == 1) {
            globalArray[i] = c1;
        } else if (i == 2) {
            globalArray[i] = c2;
        } else if (i == 3) {
            globalArray[i] = c3;
        } else {
            globalArray[i] = c4;
        }
    }

    i = 0;
    for (i = 0; i < size; i = i + 1) {
        printf("%d ", globalArray[i]);
    }

    return 0;
}
