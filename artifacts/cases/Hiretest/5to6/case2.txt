const int SIZE = 10;
int numbers[SIZE];
int sum = 0;
void initializeArray() {
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        numbers[i] = i + 1;
    }
}
int computeSum() {
    int localSum = 0;
    int i = 0;
    for (i = 0; i < SIZE; i = i + 1) {
        localSum = localSum + numbers[i];
    }
    return localSum;
}
void printSum(int total) {
    printf("Sum of array elements: %d\n", total);
}
int main(){
    initializeArray();
    sum = computeSum();
    printSum(sum);
    return 0;
}
