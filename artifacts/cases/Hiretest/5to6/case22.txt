const int SIZE = 5;
int array[SIZE] = {0, 0, 0, 0, 0};
void initializeArray(int arr[], int len, int value) {
    int i = 0;
    for (i = 0; i < len; i = i + 1) {
        arr[i] = value;
    }
}
void printArray(int arr[], int len) {
    int i = 0;
    for (i = 0; i < len; i = i + 1) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}
int main(){
    int value = getint();
    initializeArray(array, SIZE, value);
    printArray(array, SIZE);
    return 0;
}
